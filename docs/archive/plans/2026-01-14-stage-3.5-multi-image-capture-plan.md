# Stage 3.5: Multi-Image Capture Implementation Plan

**Date:** 2026-01-14
**Status:** Approved
**Depends On:** Stage 3.1 (Schema), Stage 3.3 (Rescan Edit Flow)
**Estimated Effort:** 3.5 weeks

## Overview

Enable users to capture multiple images of a single object from different angles (front, back, bottom, close-up) to improve Gemini's identification accuracy. Especially valuable for electronics with serial numbers, model info, and ports on different sides.

## Goals

- Capture 1-4 images per item from different angles
- Link all images to a single Item in the catalog
- Send all images to Gemini in a single request for comprehensive analysis
- Capture photo metadata for each image (fraud prevention)
- Store Live Photo motion clips for authenticity verification

## Non-Goals (Post-MVP)

- Automatic fingerprint-based grouping across captures
- ML-powered angle detection
- Subcollection storage schema
- Merge multiple detected objects into one item

```swift
// New state for multi-capture mode
@Published public var isMultiCaptureMode: Bool = false
@Published public var selectedObjectForMultiCapture: DetectedObject? = nil
@Published public var capturedAngles: [ImageAngle: CapturedImage] = [:]
@Published public var currentAngleGuide: ImageAngle = .front

public struct CapturedImage {
    let pixelBuffer: CVPixelBuffer
    let boundingBox: CGRect
    let metadata: PhotoMetadata
    let isLivePhoto: Bool
    let motionData: Data?
}

// Capture flow
func startMultiCapture(for object: DetectedObject) {
    isMultiCaptureMode = true
    selectedObjectForMultiCapture = object
    capturedAngles = [:]
    currentAngleGuide = .front
}

func captureCurrentAngle(pixelBuffer: CVPixelBuffer) async {
    guard let object = selectedObjectForMultiCapture else { return }

    let metadata = await photoMetadataExtractor.extractMetadata(from: pixelBuffer)

    let captured = CapturedImage(
        pixelBuffer: pixelBuffer,
        boundingBox: object.boundingBox,
        metadata: metadata,
        isLivePhoto: false,  // Detect from PHAsset if available
        motionData: nil
    )

    await MainActor.run {
        capturedAngles[currentAngleGuide] = captured
    }
}

func advanceToNextAngle() {
    let allAngles: [ImageAngle] = [.front, .back, .bottom, .closeup]
    if let currentIndex = allAngles.firstIndex(of: currentAngleGuide),
       currentIndex + 1 < allAngles.count {
        currentAngleGuide = allAngles[currentIndex + 1]
    }
}

func finishMultiCapture() async {
    guard capturedAngles.count > 0 else { return }
    await uploadMultipleImages()

    isMultiCaptureMode = false
    selectedObjectForMultiCapture = nil
    capturedAngles = [:]
}

private func uploadMultipleImages() async {
    guard let object = selectedObjectForMultiCapture,
          let storageService = storageService,
          let itemService = itemService,
          let userId = Auth.auth().currentUser?.uid else { return }

    var imageUrls: [String] = []
    var imageLabels: [String] = []
    var metadataList: [PhotoMetadata] = []

    for (angle, captured) in capturedAngles.sorted(by: { $0.key.rawValue < $1.key.rawValue }) {
        do {
            let croppedImage = try PixelBufferCropper.cropImage(
                from: captured.pixelBuffer,
                boundingBox: captured.boundingBox
            )

            let url = try await storageService.uploadCroppedObject(
                croppedImage,
                itemId: object.id.uuidString,
                userId: userId,
                angle: angle
            )

            imageUrls.append(url.absoluteString)
            imageLabels.append(angle.rawValue)
            metadataList.append(captured.metadata)

            // Upload motion clip if Live Photo
            if captured.isLivePhoto, let motionData = captured.motionData {
                try await storageService.uploadMotionClip(
                    motionData,
                    itemId: object.id.uuidString,
                    userId: userId,
                    angle: angle
                )
            }
        } catch {
            logger.error("Failed to upload \(angle.rawValue): \(error)")
        }
    }

    // Create item with all images
    do {
        try await itemService.createItemWithMultipleImages(
            itemId: object.id.uuidString,
            userId: userId,
            imageUrls: imageUrls,
            imageLabels: imageLabels,
            photoMetadataList: metadataList
        )
    } catch {
        logger.error("Failed to create multi-image item: \(error)")
    }
}
```

### Phase 3: Cloud Functions Multi-Image (1 week)

#### 3.1 Update Gemini Service

**File:** `functions/src/ai-pipeline/gemini/gemini-service.ts`

```typescript
export async function processItemWithGemini(
  imageUrls: string[]  // Now accepts array
): Promise<CatalogItem> {
  const ai = createVertexAIClient();

  // Fetch all images
  const imageDataList = await Promise.all(
    imageUrls.map(url => fetchImageBase64(url))
  );

  // Build multi-image parts
  const parts: Part[] = [
    {
      text: imageUrls.length > 1
        ? MULTI_IMAGE_PROMPT
        : SINGLE_IMAGE_PROMPT
    }
  ];

  // Add each image with angle context
  const angleLabels = ['front', 'back', 'bottom', 'closeup'];
  for (let i = 0; i < imageDataList.length; i++) {
    if (imageUrls.length > 1) {
      parts.push({
        text: `[${angleLabels[i] || 'additional'} view]`
      });
    }
    parts.push({
      inlineData: {
        mimeType: 'image/jpeg',
        data: imageDataList[i]
      }
    });
  }

  const contents: Content[] = [{
    role: 'user',
    parts
  }];

  // Rest of Gemini flow unchanged...
  const response = await ai.models.generateContent({
    model: GEMINI_MODEL_ID,
    contents,
    config: {
      systemInstruction: imageUrls.length > 1
        ? MULTI_IMAGE_SYSTEM_PROMPT
        : SYSTEM_PROMPT,
      tools: TOOLS,
      ...GENERATION_CONFIG
    }
  });

  return parseResponse(response);
}
```

#### 3.3 Update Firestore Trigger

**File:** `functions/src/triggers/onItemCreated.ts`

```typescript
exports.onItemCreated = functions.firestore
  .document('items/{itemId}')
  .onCreate(async (snap, context) => {
    const item = snap.data();

    // Determine image URLs (support both single and multi)
    let imageUrls: string[];
    if (item.imageUrls && Array.isArray(item.imageUrls)) {
      imageUrls = item.imageUrls;
    } else if (item.imageUrl) {
      imageUrls = [item.imageUrl];  // Legacy single image
    } else {
      console.error('No images found for item:', context.params.itemId);
      return;
    }

    try {
      const result = await processItemWithGemini(imageUrls);

      // Update item with catalog data
      await snap.ref.update({
        ...flattenCatalogItem(result),
        status: 'complete',
        updatedAt: admin.firestore.FieldValue.serverTimestamp()
      });
    } catch (error) {
      console.error('Gemini processing failed:', error);
      await snap.ref.update({
        status: 'failed',
        processingNotes: error.message,
        updatedAt: admin.firestore.FieldValue.serverTimestamp()
      });
    }
  });
```

## Rollback Strategy

### If Issues Discovered

1. **Disable multi-capture UI** - Set feature flag to false
2. **Single-image fallback** - System still accepts `imageUrl` (singular)
3. **Cloud Function backward compatible** - Handles both array and string

### Data Migration Not Required

- New items use `imageUrls: [String]`
- Old items keep `imageUrl: String`
- Computed property provides backward compatibility
