import UIKit
import Vision
import CoreML

/// Service for detecting objects in images using Core ML + Vision Framework
@MainActor
public final class ObjectDetectionService {

    // MARK: - Properties

    private(set) var visionModel: VNCoreMLModel?
    private var detectionRequest: VNCoreMLRequest?

    // MARK: - Initialization

    public init() {
        setupModel()
    }

    // MARK: - Public API

    /// Detect objects in the given image
    /// - Parameter image: The UIImage to analyze
    /// - Returns: Array of detected objects with bounding boxes and classifications
    public func detectObjects(in image: UIImage) async throws -> [DetectedObject] {
        guard let detectionRequest = detectionRequest else {
            throw DetectionError.modelNotLoaded
        }

        // Preprocess: Downscale to 1024x1024 for faster detection
        let processedImage = image.resized(to: CGSize(width: 1024, height: 1024))

        guard let cgImage = processedImage.cgImage else {
            throw DetectionError.invalidImage
        }

        return try await withCheckedThrowingContinuation { continuation in
            // Execute Vision request
            let handler = VNImageRequestHandler(cgImage: cgImage, options: [:])

            do {
                try handler.perform([detectionRequest])

                guard let results = detectionRequest.results as? [VNRecognizedObjectObservation] else {
                    continuation.resume(returning: [])
                    return
                }

                // Convert Vision observations to app model
                let detectedObjects = results
                    .filter { $0.confidence > 0.5 } // Confidence threshold
                    .map { observation -> DetectedObject in
                        let topLabel = observation.labels.first!
                        return DetectedObject(
                            boundingBox: observation.boundingBox,
                            className: topLabel.identifier,
                            confidence: topLabel.confidence
                        )
                    }

                continuation.resume(returning: detectedObjects)
            } catch {
                continuation.resume(throwing: error)
            }
        }
    }

    // MARK: - Private Methods

    private func setupModel() {
        do {
            // Load YOLOv3 Tiny model (bundled in app)
            let config = MLModelConfiguration()
            config.computeUnits = .all // Use Neural Engine + GPU + CPU

            let model = try YOLOv3Tiny(configuration: config)
            visionModel = try VNCoreMLModel(for: model.model)

            // Create reusable detection request
            detectionRequest = VNCoreMLRequest(model: visionModel!)
            detectionRequest?.imageCropAndScaleOption = .scaleFill

        } catch {
            print("Failed to load Core ML model: \(error)")
            visionModel = nil
            detectionRequest = nil
        }
    }
}

// MARK: - Models

/// Represents a detected object in an image
public struct DetectedObject: Equatable {
    /// Normalized bounding box [0-1] (lower-left origin, iOS coordinate space)
    public let boundingBox: CGRect

    /// Object class name (e.g., "chair", "person", "laptop")
    public let className: String

    /// Confidence score [0-1]
    public let confidence: Float

    public init(boundingBox: CGRect, className: String, confidence: Float) {
        self.boundingBox = boundingBox
        self.className = className
        self.confidence = confidence
    }
}

// MARK: - Errors

public enum DetectionError: Error, LocalizedError {
    case modelNotLoaded
    case invalidImage

    public var errorDescription: String? {
        switch self {
        case .modelNotLoaded:
            return "Core ML model failed to load"
        case .invalidImage:
            return "Invalid image format"
        }
    }
}

// MARK: - Extensions

extension UIImage {
    func resized(to size: CGSize) -> UIImage {
        UIGraphicsBeginImageContextWithOptions(size, false, 1.0)
        draw(in: CGRect(origin: .zero, size: size))
        let resizedImage = UIGraphicsGetImageFromCurrentImageContext()!
        UIGraphicsEndImageContext()
        return resizedImage
    }
}

import UIKit
import Vision

/// Service for cropping detected objects from images
public final class ImageCroppingService {

    /// Padding percentage to add around bounding boxes (0.1 = 10%)
    private let paddingPercentage: CGFloat = 0.1

    public init() {}

    /// Crop detected objects from the original high-resolution image
    /// - Parameters:
    ///   - image: Original high-res image
    ///   - detectedObjects: Objects detected by ObjectDetectionService
    /// - Returns: Array of cropped images (one per object)
    public func cropObjects(from image: UIImage, detectedObjects: [DetectedObject]) -> [UIImage] {
        guard let cgImage = image.cgImage else { return [] }

        let imageWidth = CGFloat(cgImage.width)
        let imageHeight = CGFloat(cgImage.height)

        return detectedObjects.compactMap { object in
            // Convert normalized bounding box to pixel coordinates
            var rect = VNImageRectForNormalizedRect(
                object.boundingBox,
                Int(imageWidth),
                Int(imageHeight)
            )

            // Add padding (10% of box size)
            let padding = min(rect.width, rect.height) * paddingPercentage
            rect = rect.insetBy(dx: -padding, dy: -padding)

            // Clamp to image bounds
            rect = rect.intersection(CGRect(x: 0, y: 0, width: imageWidth, height: imageHeight))

            // Crop
            guard let croppedCGImage = cgImage.cropping(to: rect) else { return nil }
            return UIImage(cgImage: croppedCGImage)
        }
    }
}

        self.bucket_name = bucket_name
        self.cloudfront_domain = cloudfront_domain
        self.max_size_mb = max_size_mb
        self.jpeg_quality = jpeg_quality

        self.s3_client = boto3.client('s3', region_name=region)
        self.max_size_bytes = int(max_size_mb * 1024 * 1024)

    def compress_image(self, image_bytes: bytes, quality: Optional[int] = None) -> bytes:
        """
        Compress image to JPEG with specified quality

        Args:
            image_bytes: Original image bytes
            quality: JPEG quality (defaults to instance quality)

        Returns:
            Compressed JPEG bytes
        """
        quality = quality or self.jpeg_quality

        # Load image
        img = Image.open(io.BytesIO(image_bytes))

        # Convert to RGB if necessary (removes alpha channel)
        if img.mode in ('RGBA', 'LA', 'P'):
            rgb_img = Image.new('RGB', img.size, (255, 255, 255))
            rgb_img.paste(img, mask=img.split()[-1] if img.mode in ('RGBA', 'LA') else None)
            img = rgb_img

        # Compress to JPEG
        buffer = io.BytesIO()
        img.save(buffer, format='JPEG', quality=quality, optimize=True)
        buffer.seek(0)

        return buffer.getvalue()

    def generate_filename(self, original_filename: str) -> str:
        """
        Generate unique filename with timestamp and UUID

        Args:
            original_filename: Original file name

        Returns:
            Unique filename: YYYYMMDD-uuid-original.jpg
        """
        timestamp = datetime.utcnow().strftime('%Y%m%d')
        unique_id = str(uuid.uuid4())[:8]

        # Extract extension
        _, ext = os.path.splitext(original_filename)
        ext = ext.lower() if ext else '.jpg'

        return f"{timestamp}-{unique_id}-{original_filename}{ext if not original_filename.endswith(ext) else ''}"

    def upload(self, image_bytes: bytes, filename: str) -> str:
        """
        Upload image to S3 and return CloudFront URL

        Args:
            image_bytes: Image data to upload
            filename: Desired filename

        Returns:
            Public CloudFront URL

        Raises:
            ValueError: If image exceeds size limit
        """
        # Validate size
        if len(image_bytes) > self.max_size_bytes:
            size_mb = len(image_bytes) / (1024 * 1024)
            raise ValueError(
                f"Image size ({size_mb:.2f} MB) exceeds maximum size ({self.max_size_mb} MB)"
            )

        # Generate unique filename
        unique_filename = self.generate_filename(filename)

        # Upload to S3
        self.s3_client.put_object(
            Bucket=self.bucket_name,
            Key=unique_filename,
            Body=image_bytes,
            ContentType='image/jpeg',
            CacheControl='public, max-age=3600',  # 1 hour cache
            Metadata={
                'uploaded-at': datetime.utcnow().isoformat(),
                'original-filename': filename
            }
        )

        # Return CloudFront URL
        return f"https://{self.cloudfront_domain}/{unique_filename}"

    def upload_with_compression(self, image_bytes: bytes, filename: str) -> str:
        """
        Compress and upload image in one step

        Args:
            image_bytes: Original image data
            filename: Desired filename

        Returns:
            Public CloudFront URL
        """
        compressed = self.compress_image(image_bytes)
        return self.upload(compressed, filename)

```python
"""
Request Queue Manager for SerpAPI Layer 2b

Manages request queuing, rate limiting, and retry logic for SerpAPI calls.
"""

import redis
import json
import time
import uuid
from typing import Dict, Any, Optional
from datetime import datetime, timedelta

class RateLimitExceeded(Exception):
    """Raised when hourly rate limit is exceeded"""
    pass

class QueueManager:
    """Redis-backed queue manager with rate limiting"""

    def __init__(
        self,
        redis_url: str = 'redis://localhost:6379',
        max_requests_per_hour: int = 3000,  # Production plan: 15K/month = 3K/hour
        queue_name: str = 'serpapi-queue',
        rate_limit_window: int = 3600  # 1 hour in seconds
    ):
        """
        Initialize queue manager

        Args:
            redis_url: Redis connection URL
            max_requests_per_hour: Maximum requests allowed per hour
            queue_name: Name of the Redis queue
            rate_limit_window: Time window for rate limiting (seconds)
        """
        self.redis_client = redis.from_url(redis_url, decode_responses=True)
        self.max_requests_per_hour = max_requests_per_hour
        self.queue_name = queue_name
        self.rate_limit_window = rate_limit_window

        # Redis keys
        self.queue_key = f"{queue_name}:pending"
        self.processing_key = f"{queue_name}:processing"
        self.failed_key = f"{queue_name}:failed"
        self.rate_limit_key = f"{queue_name}:rate_limit"

    def enqueue(self, payload: Dict[str, Any]) -> str:
        """
        Add request to queue

        Args:
            payload: Request data (image_url, search params, etc.)

        Returns:
            Request ID
        """
        request_id = str(uuid.uuid4())

        request_data = {
            'id': request_id,
            'payload': payload,
            'enqueued_at': datetime.utcnow().isoformat(),
            'attempts': 0
        }

        # Add to queue (FIFO)
        self.redis_client.rpush(self.queue_key, json.dumps(request_data))

        return request_id

    def dequeue(self) -> Optional[Dict[str, Any]]:
        """
        Get next request from queue

        Returns:
            Request data or None if queue is empty
        """
        # Check rate limit before dequeuing
        try:
            self.check_rate_limit()
        except RateLimitExceeded:
            return None

        # Move from pending to processing (atomic)
        request_json = self.redis_client.rpoplpush(self.queue_key, self.processing_key)

        if request_json is None:
            return None

        return json.loads(request_json)

    def check_rate_limit(self):
        """
        Check if we've exceeded hourly rate limit

        Raises:
            RateLimitExceeded: If limit is exceeded
        """
        current_count = self.redis_client.get(self.rate_limit_key)

        if current_count and int(current_count) >= self.max_requests_per_hour:
            raise RateLimitExceeded(
                f"Rate limit exceeded: {current_count}/{self.max_requests_per_hour} requests in last hour"
            )

    def record_request(self):
        """Record that a request was made (for rate limiting)"""
        # Increment counter with expiry
        pipe = self.redis_client.pipeline()
        pipe.incr(self.rate_limit_key)
        pipe.expire(self.rate_limit_key, self.rate_limit_window)
        pipe.execute()

    def calculate_retry_delay(self, attempt: int, base_delay: int = 1) -> int:
        """
        Calculate exponential backoff delay

        Args:
            attempt: Retry attempt number (1, 2, 3, ...)
            base_delay: Base delay in seconds

        Returns:
            Delay in seconds (1, 2, 4, 8, 16, ...)
        """
        return base_delay * (2 ** (attempt - 1))

    def mark_complete(self, request_id: str):
        """
        Mark request as complete and remove from processing

        Args:
            request_id: Request ID to complete
        """
        # Remove from processing list
        self._remove_from_processing(request_id)

    def mark_failed(self, request_id: str, error: str, max_attempts_reached: bool = False):
        """
        Mark request as failed

        Args:
            request_id: Request ID
            error: Error message
            max_attempts_reached: Whether max retry attempts were reached
        """
        # Remove from processing
        self._remove_from_processing(request_id)

        # Add to failed set with metadata
        failed_data = {
            'request_id': request_id,
            'error': error,
            'failed_at': datetime.utcnow().isoformat(),
            'max_attempts_reached': max_attempts_reached
        }
        self.redis_client.sadd(self.failed_key, json.dumps(failed_data))

    def retry_request(self, request_data: Dict[str, Any]):
        """
        Retry a failed request with incremented attempt count

        Args:
            request_data: Original request data
        """
        # Increment attempts
        request_data['attempts'] += 1

        # Calculate delay
        delay = self.calculate_retry_delay(request_data['attempts'])

        # Re-add to queue after delay
        # (In production, use Redis sorted set with timestamp for delayed retry)
        time.sleep(delay)
        self.redis_client.rpush(self.queue_key, json.dumps(request_data))

    def queue_size(self) -> int:
        """Get number of pending requests"""
        return self.redis_client.llen(self.queue_key)

    def processing_size(self) -> int:
        """Get number of requests being processed"""
        return self.redis_client.llen(self.processing_key)

    def is_failed(self, request_id: str) -> bool:
        """Check if request is in failed set"""
        # Check if any failed item has this request_id
        failed_items = self.redis_client.smembers(self.failed_key)
        for item_json in failed_items:
            item = json.loads(item_json)
            if item['request_id'] == request_id:
                return True
        return False

    def clear(self):
        """Clear all queues (for testing)"""
        self.redis_client.delete(self.queue_key)
        self.redis_client.delete(self.processing_key)
        self.redis_client.delete(self.failed_key)
        self.redis_client.delete(self.rate_limit_key)

    def _remove_from_processing(self, request_id: str):
        """Remove request from processing list by ID"""
        processing_items = self.redis_client.lrange(self.processing_key, 0, -1)
        for item_json in processing_items:
            item = json.loads(item_json)
            if item['id'] == request_id:
                self.redis_client.lrem(self.processing_key, 1, item_json)
                break
```

"""
SerpAPI Google Lens Client

Handles visual product search via SerpAPI with confidence scoring.
"""

from typing import Dict, Any, List, Optional
from serpapi import GoogleSearch

class SerpAPIError(Exception):
    """Raised when SerpAPI returns an error"""
    pass

class SerpAPIClient:
    """Client for SerpAPI Google Lens visual search"""

    def __init__(self, api_key: str):
        """
        Initialize SerpAPI client

        Args:
            api_key: SerpAPI API key
        """
        self.api_key = api_key

    def build_search_params(
        self,
        image_url: str,
        search_type: str = 'visual_matches',
        country: str = 'us',
        language: str = 'en'
    ) -> Dict[str, str]:
        """
        Build SerpAPI search parameters

        Args:
            image_url: Public URL of the image to search
            search_type: Type of search (visual_matches, exact_matches, products, all)
            country: Country code for localization
            language: Language code

        Returns:
            Dictionary of search parameters
        """
        return {
            'api_key': self.api_key,
            'engine': 'google_lens',
            'url': image_url,
            'type': search_type,
            'country': country,
            'hl': language
        }

    def search(
        self,
        image_url: str,
        search_type: str = 'visual_matches',
        country: str = 'us'
    ) -> Dict[str, Any]:
        """
        Perform visual search via SerpAPI

        Args:
            image_url: Public URL of the image
            search_type: Type of search
            country: Country code

        Returns:
            SerpAPI response dictionary

        Raises:
            SerpAPIError: If search fails
        """
        params = self.build_search_params(image_url, search_type, country)

        search = GoogleSearch(params)
        results = search.get_dict()

        # Check for errors
        if 'error' in results:
            raise SerpAPIError(results['error'])

        return results

    def parse_visual_matches(self, response: Dict[str, Any]) -> List[Dict[str, Any]]:
        """
        Parse visual matches from SerpAPI response

        Args:
            response: SerpAPI response dictionary

        Returns:
            List of parsed product matches

        Raises:
            SerpAPIError: If response contains error
        """
        # Check for errors
        if 'error' in response:
            # Empty results are not errors, return empty list
            if "hasn't returned any results" in response['error']:
                return []
            raise SerpAPIError(response['error'])

        # Extract visual matches
        visual_matches = response.get('visual_matches', [])

        # Parse each match
        parsed_matches = []
        for match in visual_matches:
            parsed = {
                'position': match['position'],
                'title': match['title'],
                'link': match['link'],
                'source': match['source'],
                'price': match.get('price', {}).get('extracted_value', 0),
                'price_string': match.get('price', {}).get('value', ''),
                'currency': match.get('price', {}).get('currency', '$'),
                'rating': match.get('rating'),
                'reviews': match.get('reviews'),
                'in_stock': match.get('in_stock'),
                'condition': match.get('condition'),
                'thumbnail': match.get('thumbnail'),
                'image': match.get('image')
            }

            # Calculate confidence proxy
            parsed['confidence'] = self.calculate_confidence_proxy(match)

            parsed_matches.append(parsed)

        return parsed_matches

    def calculate_confidence_proxy(self, match: Dict[str, Any]) -> float:
        """
        Calculate confidence score from multiple signals

        Since SerpAPI doesn't provide confidence scores, we estimate based on:
        - Position in results (lower = higher confidence)
        - Presence of price data
        - Number of reviews
        - Rating score

        Args:
            match: Single visual match dictionary

        Returns:
            Confidence score between 0.0 and 1.0
        """
        score = 0.0

        # Position signal (30%)
        position = match.get('position', 99)
        if position <= 5:
            score += 0.30
        elif position <= 10:
            score += 0.20
        elif position <= 20:
            score += 0.10

        # Price signal (30%)
        price = match.get('price', {}).get('extracted_value', 0)
        if price > 0:
            score += 0.30

        # Reviews signal (20%)
        reviews = match.get('reviews', 0)
        if reviews and reviews > 100:
            score += 0.20
        elif reviews and reviews > 10:
            score += 0.10

        # Rating signal (20%)
        rating = match.get('rating', 0)
        if rating and rating >= 4.0:
            score += 0.20
        elif rating and rating >= 3.0:
            score += 0.10

        return min(score, 1.0)

    def filter_by_confidence(
        self,
        matches: List[Dict[str, Any]],
        min_confidence: float = 0.5
    ) -> List[Dict[str, Any]]:
        """
        Filter matches by minimum confidence threshold

        Args:
            matches: List of parsed matches
            min_confidence: Minimum confidence score (0.0 to 1.0)

        Returns:
            Filtered list of matches
        """
        return [m for m in matches if m['confidence'] >= min_confidence]

    def get_top_matches(
        self,
        matches: List[Dict[str, Any]],
        top_n: int = 5
    ) -> List[Dict[str, Any]]:
        """
        Get top N matches sorted by confidence

        Args:
            matches: List of parsed matches
            top_n: Number of top matches to return

        Returns:
            Top N matches sorted by confidence (descending)
        """
        sorted_matches = sorted(matches, key=lambda m: m['confidence'], reverse=True)
        return sorted_matches[:top_n]

```python
"""
Brand/Model Parser using Claude Haiku

Extracts structured product data from combined title strings.
"""

import os
import json
from typing import Optional, List, Dict, Any
from dataclasses import dataclass
from anthropic import Anthropic

@dataclass
class ParsedProduct:
    """Structured product data"""
    brand: str
    model: str
    variant: Optional[str] = None
    category: Optional[str] = None
    confidence: float = 1.0

class BrandParser:
    """Parser for extracting brand/model from product titles using Claude Haiku"""

    def __init__(
        self,
        api_key: str,
        model: str = 'claude-haiku-4-20250514',
        brand_database_path: str = 'brand_database.json'
    ):
        """
        Initialize brand parser

        Args:
            api_key: Anthropic API key
            model: Claude model ID (Haiku for cost efficiency)
            brand_database_path: Path to JSON file with known brands
        """
        self.client = Anthropic(api_key=api_key)
        self.model = model
        self.brand_database_path = brand_database_path
        self.known_brands = self.load_brand_database()

    def load_brand_database(self) -> List[str]:
        """
        Load known brands from JSON database

        Returns:
            List of brand names
        """
        try:
            with open(self.brand_database_path, 'r') as f:
                database = json.load(f)

            # Flatten all categories into single list
            all_brands = []
            for category_brands in database.values():
                all_brands.extend(category_brands)

            return list(set(all_brands))  # Remove duplicates
        except FileNotFoundError:
            # Fallback to empty list if database not found
            return []

    def is_known_brand(self, brand: str) -> bool:
        """
        Check if brand exists in database

        Args:
            brand: Brand name to check

        Returns:
            True if brand is known
        """
        return brand in self.known_brands

    def build_prompt(self, title: str) -> str:
        """
        Build LLM prompt for parsing

        Args:
            title: Product title string

        Returns:
            Prompt string
        """
        return f"""Parse this product title into structured data:

"{title}"

Extract:
- brand: Manufacturer/brand name
- model: Product model name/number
- variant: Specific variant (color, size, version) if mentioned
- category: Product category

Return ONLY valid JSON with these exact keys. Be concise.

Example:
{{"brand": "Sony", "model": "DualSense", "variant": "Midnight Black", "category": "Gaming Controllers"}}"""

    def parse(self, title: str) -> ParsedProduct:
        """
        Parse product title into structured data

        Args:
            title: Product title string (e.g., "Sony DualSense Wireless Controller - Midnight Black")

        Returns:
            ParsedProduct with extracted fields
        """
        # Build prompt
        prompt = self.build_prompt(title)

        # Call Claude Haiku
        response = self.client.messages.create(
            model=self.model,
            max_tokens=200,
            messages=[{
                "role": "user",
                "content": prompt
            }]
        )

        # Parse JSON response
        response_text = response.content[0].text.strip()

        # Remove markdown code blocks if present
        if response_text.startswith('```'):
            response_text = response_text.split('\n', 1)[1]
            response_text = response_text.rsplit('```', 1)[0]

        parsed_json = json.loads(response_text)

        # Create ParsedProduct
        result = ParsedProduct(
            brand=parsed_json.get('brand', ''),
            model=parsed_json.get('model', ''),
            variant=parsed_json.get('variant'),
            category=parsed_json.get('category'),
            confidence=1.0 if self.is_known_brand(parsed_json.get('brand', '')) else 0.8
        )

        return result

    def parse_batch(self, titles: List[str]) -> List[ParsedProduct]:
        """
        Parse multiple titles in batch

        Args:
            titles: List of product titles

        Returns:
            List of ParsedProduct objects
        """
        return [self.parse(title) for title in titles]
```

import UIKit
import GoogleGenerativeAI

/// Service for extracting product attributes using Gemini 2.5 Flash-Lite
@MainActor
public final class GeminiAttributeService {

    // MARK: - Properties

    private let model: GenerativeModel
    private let apiKey: String

    // MARK: - Initialization

    public init(apiKey: String, modelID: String = "gemini-2.5-flash-lite") {
        self.apiKey = apiKey
        self.model = GenerativeModel(name: modelID, apiKey: apiKey)
    }

    // MARK: - Public API

    /// Extract attributes from product image
    /// - Parameter image: Product image
    /// - Returns: Structured product attributes
    public func extractAttributes(from image: UIImage) async throws -> ProductAttributes {
        // Convert image to Data
        guard let imageData = image.jpegData(compressionQuality: 0.85) else {
            throw AttributeError.invalidImage
        }

        // Build prompt
        let prompt = buildPrompt()

        // Create multimodal content
        let imagePart = ModelContent.Part.data(mimetype: "image/jpeg", imageData)
        let textPart = ModelContent.Part.text(prompt)

        // Generate response
        let response = try await model.generateContent(textPart, imagePart)

        guard let responseText = response.text else {
            throw AttributeError.emptyResponse
        }

        // Parse JSON response
        return try parseResponse(responseText)
    }

    // MARK: - Internal Methods

    func buildPrompt() -> String {
        return """
        Analyze this product image and extract:
        - condition: "new", "used", or "damaged"
        - color: primary color name
        - material: material type
        - category: product category
        - size: estimated size category ("small", "medium", "large")
        - features: list of notable features (max 5)
        - confidence: your confidence in these assessments (0.0 to 1.0)

        Return ONLY valid JSON with these exact keys. Be concise.
        """
    }

    func parseResponse(_ jsonString: String) throws -> ProductAttributes {
        // Remove markdown code blocks if present
        var cleanJSON = jsonString.trimmingCharacters(in: .whitespacesAndNewlines)
        if cleanJSON.hasPrefix("```") {
            // Remove first line and last line
            let lines = cleanJSON.components(separatedBy: .newlines)
            cleanJSON = lines.dropFirst().dropLast().joined(separator: "\n")
        }

        guard let data = cleanJSON.data(using: .utf8) else {
            throw AttributeError.invalidJSON
        }

        let decoder = JSONDecoder()
        return try decoder.decode(ProductAttributes.self, from: data)
    }
}

    public let condition: String
    public let color: String
    public let material: String
    public let category: String
    public let size: String?
    public let features: [String]?
    public let confidence: Double

    public init(
        condition: String,
        color: String,
        material: String,
        category: String,
        size: String? = nil,
        features: [String]? = nil,
        confidence: Double
    ) {
        self.condition = condition
        self.color = color
        self.material = material
        self.category = category
        self.size = size
        self.features = features
        self.confidence = confidence
    }
}

// MARK: - Errors

public enum AttributeError: Error, LocalizedError {
    case invalidImage
    case emptyResponse
    case invalidJSON

    public var errorDescription: String? {
        switch self {
        case .invalidImage:
            return "Failed to convert image to JPEG data"
        case .emptyResponse:
            return "Gemini returned empty response"
        case .invalidJSON:
            return "Failed to parse JSON response"
        }
    }
}

/// Orchestrates the complete 4-layer catalog pipeline
@MainActor
public final class CatalogPipeline {

    private let configuration: PipelineConfiguration

    // Layer services
    private let detectionService: ObjectDetectionService
    private let croppingService: ImageCroppingService
    private let geminiService: GeminiAttributeService
    // Note: SerpAPI and Claude services would be backend calls

    public init(configuration: PipelineConfiguration) {
        self.configuration = configuration

        // Initialize services
        self.detectionService = ObjectDetectionService()
        self.croppingService = ImageCroppingService()
        self.geminiService = GeminiAttributeService(apiKey: configuration.geminiAPIKey)
    }

    /// Process image through complete pipeline
    public func process(_ image: UIImage) async throws -> CatalogEntry {
        // Layer 1: Object Detection + Cropping
        let detectedObjects = try await runLayer1Detection(image)
        let crops = croppingService.cropObjects(from: image, detectedObjects: detectedObjects)

        // Process each detected object through remaining layers
        var processedObjects: [CatalogEntry.DetectedProductObject] = []

        for (index, detectedObject) in detectedObjects.enumerated() {
            guard index < crops.count else { continue }

            let croppedImage = crops[index]

            // Layer 2a: Attribute Extraction (parallel)
            // Layer 2b: Product Search (parallel)
            async let attributes = runLayer2aAttributes(croppedImage)
            async let productMatches = runLayer2bProductSearch(croppedImage)

            let (attrs, matches) = try await (attributes, productMatches)

            // Parse brand/model from top match
            var parsedBrand: String?
            var parsedModel: String?
            if let topMatch = matches.first {
                let parsed = try await parseBrandModel(topMatch.title)
                parsedBrand = parsed.brand
                parsedModel = parsed.model
            }

            // Layer 3: AI Synthesis (TODO: implement)
            let finalMetadata = synthesizeMetadata(
                attributes: attrs,
                matches: matches,
                brand: parsedBrand,
                model: parsedModel
            )

            let processedObject = CatalogEntry.DetectedProductObject(
                boundingBox: detectedObject.boundingBox,
                className: detectedObject.className,
                croppedImage: croppedImage,
                attributes: attrs,
                productMatches: matches,
                parsedBrand: parsedBrand,
                parsedModel: parsedModel,
                finalMetadata: finalMetadata
            )

            processedObjects.append(processedObject)
        }

        return CatalogEntry(
            originalImage: image,
            detectedObjects: processedObjects,
            processedAt: Date()
        )
    }

    // MARK: - Layer Functions

    func runLayer1Detection(_ image: UIImage) async throws -> [DetectedObject] {
