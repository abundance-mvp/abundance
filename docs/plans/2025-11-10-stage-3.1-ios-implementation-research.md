## Verified Technical Constraints

### Swift 6 Strict Concurrency

**Constraint**: All code must compile with Swift 6 strict concurrency enabled (`SWIFT_STRICT_CONCURRENCY = complete`).

**Implications**:
- ViewModels must be annotated with `@MainActor` to access `@Published` properties
- Repository methods must use `async/await` (no completion handlers)
- Vision Framework processing must use `Task.detached` to avoid blocking main thread

**Reference**: RESEARCH-VALIDATION-stage-3.1.md, Claim 1

### Firebase iOS SDK Partial Swift 6 Support

**Constraint**: Firebase iOS SDK 11.11.0+ supports async/await but has strict concurrency warnings.

**Workaround**:
```swift
@preconcurrency import FirebaseAuth
@preconcurrency import FirebaseFirestore
@preconcurrency import FirebaseStorage
```

**Timeline**: Full Swift 6 strict concurrency support planned for Firebase iOS SDK 12.x (Q1 2026).

**Reference**: RESEARCH-VALIDATION-stage-3.1.md, Claim 4

### SwiftUI @Observable Preference

**Constraint**: SwiftUI 6 `@Observable` macro is 30-50% faster than Combine's `@Published`.

**Recommendation**: Use `@Observable` for new ViewModels, migrate from `@Published` incrementally.

**Migration Example**:
```swift
// Before (Combine)
@MainActor
class CatalogViewModel: ObservableObject {
    @Published var items: [CatalogItem] = []
}

// After (@Observable)
@Observable
@MainActor
class CatalogViewModel {
    var items: [CatalogItem] = []
}
```

**Reference**: RESEARCH-VALIDATION-stage-3.1.md, Claim 2

### Vision Framework Concurrency Pattern

**Constraint**: `VNImageRequestHandler.perform(_:)` is synchronous and CPU-intensive.

**Pattern**: Use `Task.detached` to run Vision requests on background thread:
```swift
Task.detached(priority: .userInitiated) {
    let handler = VNImageRequestHandler(cgImage: image.cgImage!)
    try handler.perform([request])

    await MainActor.run {
        // Update UI with results
    }
}
```

**Reference**: RESEARCH-VALIDATION-stage-3.1.md, Claim 5

### ConcentricRectangle iOS 26+ Availability

**Constraint**: `ConcentricRectangle` shape is available iOS 26.0+ only.

**Fallback Pattern**:
```swift
if #available(iOS 26, *) {
    content.background(.thickMaterial, in: ConcentricRectangle(cornerRadius: 16, inset: 0))
} else {
    content.background(.thickMaterial, in: RoundedRectangle(cornerRadius: 16))
}
```

**Reference**: RESEARCH-VALIDATION-stage-3.1.md, Claim 3

**CatalogViewModel (Production-Ready)**:
```swift
import SwiftUI
import FirebaseFirestore
import Combine

@Observable
@MainActor
class CatalogViewModel {
    // MARK: - Published State
    var items: [CatalogItem] = []
    var filteredItems: [CatalogItem] = []
    var searchText: String = "" {
        didSet { filterItems() }
    }
    var isLoading: Bool = false
    var error: Error?

    // MARK: - Dependencies
    private let repository: CatalogRepository
    private var listenerTask: Task<Void, Never>?

    // MARK: - Initialization
    init(repository: CatalogRepository) {
        self.repository = repository
    }

    // MARK: - Public Methods
    func startListening() {
        listenerTask = Task {
            do {
                for try await items in repository.observeItems() {
                    self.items = items
                    filterItems()
                }
            } catch {
                self.error = error
            }
        }
    }

    func stopListening() {
        listenerTask?.cancel()
    }

    func deleteItem(_ item: CatalogItem) async throws {
        try await repository.deleteItem(item.id)
    }

    // MARK: - Private Methods
    private func filterItems() {
        if searchText.isEmpty {
            filteredItems = items
        } else {
            filteredItems = items.filter {
                $0.name.localizedCaseInsensitiveContains(searchText) ||
                $0.category.localizedCaseInsensitiveContains(searchText)
            }
        }
    }
}
```

           let authResult = try await Auth.auth().signIn(with: credential)
           return authResult.user
       }

       }

       }

2. **Firestore Real-Time Listener with AsyncStream**
   ```swift
   @preconcurrency import FirebaseFirestore

   actor FirestoreCatalogRepository: CatalogRepository {
       private let db = Firestore.firestore()

       func observeItems() -> AsyncThrowingStream<[CatalogItem], Error> {
           AsyncThrowingStream { continuation in
               let listener = db.collection("items")
                   .addSnapshotListener { snapshot, error in
                       if let error = error {
                           continuation.finish(throwing: error)
                       } else if let snapshot = snapshot {
                           let items = snapshot.documents.compactMap { doc in
                               try? doc.data(as: CatalogItem.self)
                           }
                           continuation.yield(items)
                       }
                   }

               continuation.onTermination = { _ in
                   listener.remove()
               }
           }
       }
   }
   ```

3. **Firebase Storage Async Upload**
   ```swift
   @preconcurrency import FirebaseStorage

   actor StorageService {
       private let storage = Storage.storage()

       func uploadImage(_ image: UIImage, path: String) async throws -> URL {
           guard let imageData = image.jpegData(compressionQuality: 0.8) else {
               throw StorageError.invalidImage
           }

           let ref = storage.reference().child(path)
           let metadata = StorageMetadata()
           metadata.contentType = "image/jpeg"

           _ = try await ref.putDataAsync(imageData, metadata: metadata)
           let downloadURL = try await ref.downloadURL()
           return downloadURL
       }
   }
   ```

### Task Group 4: Vision Framework Integration

**Document**: `docs/design/CODE-EXAMPLE-004-vision-framework-patterns.md`

**Objective**: Show how to use Vision Framework with Swift 6 concurrency (Task.detached pattern).

**Code Examples to Include**:

1. **VNCoreMLRequest with Task.detached**
   ```swift
   @MainActor
   class CameraViewModel: ObservableObject {
       @Published var detectedObjects: [DetectedObject] = []
       @Published var isProcessing: Bool = false

       func processImage(_ image: UIImage) async {
           isProcessing = true
           defer { isProcessing = false }

           do {
               let objects = try await detectObjects(in: image)
               detectedObjects = objects
           } catch {
               print("Vision error: \(error)")
           }
       }

       private func detectObjects(in image: UIImage) async throws -> [DetectedObject] {
           try await Task.detached(priority: .userInitiated) {
               guard let cgImage = image.cgImage else {
                   throw VisionError.invalidImage
               }

               let model = try VNCoreMLModel(for: YOLOv3Tiny().model)
               let request = VNCoreMLRequest(model: model)
               let handler = VNImageRequestHandler(cgImage: cgImage)
               try handler.perform([request])

               let results = (request.results as? [VNRecognizedObjectObservation]) ?? []
               return results.map { observation in
                   DetectedObject(
                       label: observation.labels.first?.identifier ?? "",
                       confidence: observation.confidence,
                       boundingBox: observation.boundingBox
                   )
               }
           }.value
       }
   }
   ```

2. **VNDetectBarcodesRequest Async Pattern**
   ```swift
   func detectBarcodes(in image: UIImage) async throws -> [String] {
       try await Task.detached(priority: .userInitiated) {
           guard let cgImage = image.cgImage else {
               throw VisionError.invalidImage
           }

           let request = VNDetectBarcodesRequest()
           request.symbologies = [.upce, .ean13, .qr]

           let handler = VNImageRequestHandler(cgImage: cgImage)
           try handler.perform([request])

           let barcodes = (request.results as? [VNBarcodeObservation]) ?? []
           return barcodes.compactMap { $0.payloadStringValue }
       }.value
   }
   ```

3. **Image Preprocessing Utilities**
   ```swift
   extension UIImage {
       func cropped(to boundingBox: CGRect) -> UIImage? {
           guard let cgImage = cgImage else { return nil }

           let width = CGFloat(cgImage.width)
           let height = CGFloat(cgImage.height)

           let rect = CGRect(
               x: boundingBox.origin.x * width,
               y: (1 - boundingBox.origin.y - boundingBox.height) * height,
               width: boundingBox.width * width,
               height: boundingBox.height * height
           )

           guard let croppedCGImage = cgImage.cropping(to: rect) else { return nil }
           return UIImage(cgImage: croppedCGImage)
       }
   }
   ```

#### TEST-EXAMPLE-001: ViewModel Unit Tests

**Structure**: Given/When/Then pattern with Swift Testing framework

**Example: CatalogViewModel Unit Test**
```swift
import Testing
@testable import CatalogFeature

@MainActor
@Suite("CatalogViewModel Tests")
struct CatalogViewModelTests {

    @Test("Fetch items success")
    func testFetchItemsSuccess() async throws {
        // Given
        let mockRepo = MockCatalogRepository(stubbedItems: [
            CatalogItem(id: "1", name: "Hammer", category: "Tools"),
            CatalogItem(id: "2", name: "Screwdriver", category: "Tools")
        ])
        let viewModel = CatalogViewModel(repository: mockRepo)

        // When
        await viewModel.fetchItems()

        // Then
        #expect(viewModel.items.count == 2)
        #expect(viewModel.items.first?.name == "Hammer")
        #expect(viewModel.isLoading == false)
        #expect(viewModel.error == nil)
    }

    @Test("Fetch items failure")
    func testFetchItemsFailure() async throws {
        // Given
        let mockRepo = MockCatalogRepository(shouldThrowError: true)
        let viewModel = CatalogViewModel(repository: mockRepo)

        // When
        await viewModel.fetchItems()

        // Then
        #expect(viewModel.items.isEmpty)
        #expect(viewModel.error != nil)
    }

    @Test("Search filters items")
    func testSearchFiltersItems() async throws {
        // Given
        let mockRepo = MockCatalogRepository(stubbedItems: [
            CatalogItem(id: "1", name: "Hammer", category: "Tools"),
            CatalogItem(id: "2", name: "Tent", category: "Camping")
        ])
        let viewModel = CatalogViewModel(repository: mockRepo)
        await viewModel.fetchItems()

        // When
        viewModel.searchText = "hammer"

        // Then
        #expect(viewModel.filteredItems.count == 1)
        #expect(viewModel.filteredItems.first?.name == "Hammer")
    }
}
```

**Acceptance Criteria**:
- Given: CatalogViewModel test suite
- When: Running `swift test`
- Then: 90%+ code coverage for ViewModel logic
- Coverage target: 80% (from TEST-STRATEGY-001)

#### TEST-EXAMPLE-002: Integration & UI Tests

**Firebase Emulator Integration Test**:
```swift
import Testing
import FirebaseFirestore
@testable import Firebase

@Suite("Firebase Integration Tests")
struct FirebaseIntegrationTests {

    init() async throws {
        // Configure Firestore Emulator
        let settings = Firestore.firestore().settings
        settings.host = "localhost:8080"
        settings.cacheSettings = MemoryCacheSettings()
        settings.isSSLEnabled = false
        Firestore.firestore().settings = settings
    }

    @Test("Create and fetch item from Firestore")
    func testCreateAndFetchItem() async throws {
        let repository = FirestoreCatalogRepository()
        let item = CatalogItem(id: UUID().uuidString, name: "Test Item", category: "Test")

        try await repository.createItem(item)
        let fetched = try await repository.fetchItem(id: item.id)

        #expect(fetched.name == "Test Item")
        #expect(fetched.category == "Test")
    }
}
```
