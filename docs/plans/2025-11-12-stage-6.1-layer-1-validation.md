#### 4a. HouseholdItemDetectorTests.swift

```swift
import XCTest
@testable import VisionCore

/// Layer 1 Validation: Household Item Detection
class HouseholdItemDetectorTests: XCTestCase {

    var sut: HouseholdItemDetector!
    var goldenDataset: GoldenDataset!

    override func setUp() async throws {
        super.setUp()

        // Load golden dataset manifest
        goldenDataset = try GoldenDataset.load()

        // Initialize detector (uses real Vision Framework)
        let visionService = VisionService()
        let barcodeService = BarcodeService()
        sut = HouseholdItemDetector(
            visionService: visionService,
            barcodeService: barcodeService
        )
    }

    /// TC-001: Household Item Detection Accuracy > 60%
    func testHouseholdItemDetectionAccuracy() async throws {
        var correctPredictions = 0
        var totalItems = 0
        var results: [AccuracyResult] = []

        for item in goldenDataset.items {
            let image = try loadImage(path: item.imagePath)
            let detections = try await sut.detectHouseholdItems(in: image)

            // Check if top detection matches ground truth category
            let predicted = detections.first?.label ?? "unknown"
            let expected = item.groundTruth.category

            let isCorrect = categoriesMatch(predicted, expected)
            if isCorrect {
                correctPredictions += 1
            }
            totalItems += 1

            results.append(AccuracyResult(
                id: item.id,
                predicted: predicted,
                expected: expected,
                correct: isCorrect
            ))
        }

        let accuracy = Double(correctPredictions) / Double(totalItems)

        // Export results to CSV
        try exportAccuracyResults(results, to: "layer1_accuracy_results.csv")

        // Assert > 60% threshold
        XCTAssertGreaterThan(accuracy, 0.60, "Household item detection accuracy (\(accuracy * 100)%) must exceed 60%")

        print("✅ Household Item Detection: \(correctPredictions)/\(totalItems) = \(accuracy * 100)%")
    }

    /// TC-002: Barcode Detection Accuracy > 95%
    func testBarcodeDetectionAccuracy() async throws {
        // Filter golden dataset to barcode-bearing images only
        let barcodeItems = goldenDataset.items.filter { $0.groundTruth.barcode != nil }

        var correctDetections = 0
        var totalBarcodes = 0

        for item in barcodeItems {
            let image = try loadImage(path: item.imagePath)
            let barcodeService = BarcodeService()
            let detectedBarcodes = try await barcodeService.detectBarcodes(in: image)

            let expected = item.groundTruth.barcode!

            if detectedBarcodes.contains(expected) {
                correctDetections += 1
            }
            totalBarcodes += 1
        }

        let accuracy = Double(correctDetections) / Double(totalBarcodes)

        // Assert > 95% threshold
        XCTAssertGreaterThan(accuracy, 0.95, "Barcode detection accuracy (\(accuracy * 100)%) must exceed 95%")

        print("✅ Barcode Detection: \(correctDetections)/\(totalBarcodes) = \(accuracy * 100)%")
    }

    /// TC-003: Processing Latency < 500ms (p90)
    func testProcessingLatency() async throws {
        var latencies: [TimeInterval] = []

        for item in goldenDataset.items {
            let image = try loadImage(path: item.imagePath)

            let startTime = Date()
            _ = try await sut.detectHouseholdItems(in: image)
            let endTime = Date()

            let latency = endTime.timeIntervalSince(startTime)
            latencies.append(latency)
        }

        // Calculate p90
        latencies.sort()
        let p90Index = Int(Double(latencies.count) * 0.9)
        let p90Latency = latencies[p90Index]

        // Assert < 500ms
        XCTAssertLessThan(p90Latency, 0.5, "p90 latency (\(p90Latency * 1000)ms) must be < 500ms")

        print("✅ p90 Latency: \(p90Latency * 1000)ms")

        // Export latency distribution
        try exportLatencyResults(latencies, to: "layer1_latency_distribution.csv")
    }

    /// TC-004: Confidence Categorization
    func testConfidenceCategorization() {
        XCTAssertEqual(ConfidenceScore(raw: 0.85).category, .high)
        XCTAssertEqual(ConfidenceScore(raw: 0.7).category, .medium)
        XCTAssertEqual(ConfidenceScore(raw: 0.5).category, .low)
    }

    /// TC-005: Household Class Filtering
    func testHouseholdClassFiltering() async throws {
        // Use image with mixed classes (person, car, backpack)
        let mixedImage = try loadImage(path: "test_mixed_classes.jpg")
        let detections = try await sut.detectHouseholdItems(in: mixedImage)

        // Verify no irrelevant classes
        let irrelevantClasses = ["person", "car", "dog", "bicycle"]
        for detection in detections {
            XCTAssertFalse(irrelevantClasses.contains(detection.label), "Detected irrelevant class: \(detection.label)")
        }
    }

    /// TC-006: Non-Maximum Suppression
    func testNonMaximumSuppression() async throws {
        // Image with overlapping backpack detections
        let overlappingImage = try loadImage(path: "test_overlapping_objects.jpg")
        let detections = try await sut.detectHouseholdItems(in: overlappingImage)

        // Should have removed duplicate detections (IoU > 0.5)
        XCTAssertLessThanOrEqual(detections.count, 2, "NMS should remove overlapping detections")
    }

    // MARK: - Helper Methods

    private func loadImage(path: String) throws -> UIImage {
        let bundle = Bundle(for: type(of: self))
        guard let image = UIImage(named: path, in: bundle, compatibleWith: nil) else {
            throw TestError.imageNotFound(path)
        }
        return image
    }

    private func categoriesMatch(_ predicted: String, _ expected: String) -> Bool {
        // Map COCO classes to golden dataset categories
        let categoryMap: [String: String] = [
            "backpack": "camping",
            "handbag": "clothing",
            "suitcase": "camping",
            "bottle": "kitchen",
            "cup": "kitchen",
            "fork": "kitchen",
            // ... (full mapping in implementation)
        ]

        return categoryMap[predicted] == expected
    }

    private func exportAccuracyResults(_ results: [AccuracyResult], to filename: String) throws {
        // CSV export logic
        let csvHeader = "id,predicted,expected,correct\n"
        let csvRows = results.map { "\($0.id),\($0.predicted),\($0.expected),\($0.correct)" }.joined(separator: "\n")
        let csvContent = csvHeader + csvRows

        // Write to test results directory
        try csvContent.write(toFile: filename, atomically: true, encoding: .utf8)
    }

    private func exportLatencyResults(_ latencies: [TimeInterval], to filename: String) throws {
        // CSV export logic
        let csvHeader = "image_index,latency_ms\n"
        let csvRows = latencies.enumerated().map { "\($0.offset),\($0.element * 1000)" }.joined(separator: "\n")
        let csvContent = csvHeader + csvRows

        try csvContent.write(toFile: filename, atomically: true, encoding: .utf8)
    }
}

struct AccuracyResult {
    let id: String
    let predicted: String
    let expected: String
    let correct: Bool
}

enum TestError: Error {
    case imageNotFound(String)
}
```

#### 4b. GoldenDataset.swift

```swift
import Foundation

/// Golden dataset loader (100 household items)
struct GoldenDataset: Codable {
    let version: String
    let created: String
    let totalItems: Int
    let items: [GoldenDatasetItem]

    static func load() throws -> GoldenDataset {
        let bundle = Bundle(for: HouseholdItemDetectorTests.self)
        guard let url = bundle.url(forResource: "golden_dataset_manifest", withExtension: "json") else {
            throw GoldenDatasetError.manifestNotFound
        }

        let data = try Data(contentsOf: url)
        let decoder = JSONDecoder()
        let wrapper = try decoder.decode(GoldenDatasetWrapper.self, from: data)
        return wrapper.golden_dataset_v1
    }
}

struct GoldenDatasetWrapper: Codable {
    let golden_dataset_v1: GoldenDataset
}

struct GoldenDatasetItem: Codable {
    let id: String
    let imagePath: String
    let groundTruth: GroundTruth

    enum CodingKeys: String, CodingKey {
        case id
        case imagePath = "image_path"
        case groundTruth = "ground_truth"
    }
}

struct GroundTruth: Codable {
    let category: String
    let name: String
    let brand: String?
    let color: String
    let material: String
    let condition: String
    let barcode: String?
    let estimatedValue: Double?
}

enum GoldenDatasetError: Error {
    case manifestNotFound
}
```

## Risk Mitigation

### Risk 1: Golden Dataset Creation Delays Stage 6.1
**Impact**: High (blocks test execution)
**Mitigation**:
- Start golden dataset curation immediately (parallel with test infrastructure setup)
- Use public dataset images as fallback (COCO, ImageNet)
- Minimum viable dataset: 50 images (reduce from 100 if time-constrained)

### Risk 2: Accuracy Below 60% Threshold
**Impact**: Medium (Sprint 2 blocker)
**Mitigation**:
- Tune confidence threshold (default 0.6 → 0.5 or 0.55)
- Review golden dataset labeling (ensure categories align with COCO classes)
- Accept conditional pass (within 5% of threshold) with mitigation plan

### Risk 3: No Access to iPhone 15 Pro Device
**Impact**: Medium (cannot test Neural Engine optimization)
**Mitigation**:
- Use iOS Simulator (acceptable for functional tests, latency will be higher)
- Document limitation in validation report
- Plan device testing for Sprint 2
