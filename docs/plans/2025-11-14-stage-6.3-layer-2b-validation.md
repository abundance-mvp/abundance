```python
def lookup_barcode(barcode: str, item_id: str) -> Dict:
    """
    Lookup barcode using UPCitemdb API

    Args:
        barcode: UPC/EAN barcode string
        item_id: Item ID for logging

    Returns:
        Dict with found (bool), product data, latency, cost
    """
    global total_cost

    print(f"[UPCitemdb] Looking up barcode: {barcode} for item {item_id}")

    start_time = time.time()

    try:
        response = requests.get(
            f"https://api.upcitemdb.com/prod/trial/lookup?upc={barcode}",
            headers={
                'Authorization': f'Bearer {UPCITEMDB_API_KEY}',
                'Accept': 'application/json'
            },
            timeout=5
        )

        latency = (time.time() - start_time) * 1000  # Convert to ms

        if response.status_code != 200:
            print(f"[UPCitemdb] ❌ API error: HTTP {response.status_code}")
            return {
                'found': False,
                'error': f'HTTP {response.status_code}',
                'latency': latency,
                'cost': 0.000165  # $0.000165 per lookup (corrected from research)
            }

        data = response.json()

        # UPCitemdb cost: $99/month ÷ 600K requests = $0.000165 per lookup
        cost = 0.000165
        total_cost += cost

        if data.get('code') != 'OK' or data.get('total', 0) == 0:
            print(f"[UPCitemdb] ⚠️ Barcode not found: {barcode}")
            return {
                'found': False,
                'error': 'not_found',
                'latency': latency,
                'cost': cost
            }

        item = data['items'][0]
        print(f"[UPCitemdb] ✅ Found: {item.get('title', 'Unknown')} ({latency:.0f}ms)")

        return {
            'found': True,
            'product': {
                'name': item.get('title'),
                'brand': item.get('brand'),
                'category': item.get('category'),
                'description': item.get('description'),
                'upc': item.get('upc'),
                'ean': item.get('ean'),
                'images': item.get('images', [])
            },
            'latency': latency,
            'cost': cost
        }

    except Exception as e:
        latency = (time.time() - start_time) * 1000
        print(f"[UPCitemdb] ❌ Exception: {e}")
        return {
            'found': False,
            'error': str(e),
            'latency': latency,
            'cost': 0.000165
        }
```

```python
def search_with_serpapi(image_url: str, item_id: str) -> Dict:
    """
    Search for product using SerpAPI Google Lens

    Args:
        image_url: Public HTTPS image URL
        item_id: Item ID for logging

    Returns:
        Dict with visual_matches, latency, cost
    """
    global serpapi_call_count, total_cost

    if serpapi_call_count >= MAX_SERPAPI_CALLS:
        raise Exception(f"SerpAPI budget exceeded ({MAX_SERPAPI_CALLS} calls)")

    print(f"[SerpAPI] Searching image: {image_url} for item {item_id}")

    start_time = time.time()

    try:
        from serpapi import GoogleSearch

        params = {
            "engine": "google_lens",
            "url": image_url,
            "api_key": SERPAPI_API_KEY
        }

        search = GoogleSearch(params)
        results = search.get_dict()

        latency = (time.time() - start_time) * 1000

        # SerpAPI cost: $0.015 per search (verified from research)
        cost = 0.015
        total_cost += cost
        serpapi_call_count += 1

        visual_matches = results.get('visual_matches', [])

        print(f"   API calls: {serpapi_call_count}/{MAX_SERPAPI_CALLS}")

        return {
            'visual_matches': visual_matches,
            'search_metadata': results.get('search_metadata', {}),
            'latency': latency,
            'cost': cost
        }

        latency = (time.time() - start_time) * 1000
```

```python
def parse_serpapi_results(visual_matches: List[Dict], item_id: str) -> Dict:
    """
    Parse SerpAPI visual matches using Claude Haiku 4.5

    Args:
        visual_matches: Array of visual match objects from SerpAPI
        item_id: Item ID for logging

    Returns:
        Dict with brand, model, variant, estimatedValue, confidence, cost
    """
    global claude_call_count, total_cost

    if claude_call_count >= MAX_CLAUDE_CALLS:
        raise Exception(f"Claude budget exceeded ({MAX_CLAUDE_CALLS} calls)")

    if not visual_matches:
        return {
            'brand': None,
            'model': None,
            'variant': None,
            'estimatedValue': None,
            'confidence': 0.0,
            'error': 'no_visual_matches',
            'cost': 0.0
        }

    print(f"[Claude] Parsing {len(visual_matches)} visual matches for item {item_id}")

    anthropic = Anthropic(api_key=ANTHROPIC_API_KEY)

    prompt = f"""You are analyzing product search results to extract structured product information.

Given these product search results, extract:
1. **brand**: The manufacturer/brand name (e.g., "Coleman", "Nike", "Apple")
2. **model**: The specific model or product line (e.g., "Triton", "Air Max", "iPhone 15")
3. **variant**: Any variant details (e.g., "2-Burner", "Pro", "128GB")
4. **estimatedValue**: The estimated market value in USD (average of listed prices, or estimate if no prices)

Product search results:
{json.dumps(visual_matches[:5], indent=2)}

Return ONLY a JSON object with this exact structure:
{{
  "brand": "...",
  "model": "...",
  "variant": "...",
  "estimatedValue": number,
  "confidence": number (0.0-1.0),
  "reasoning": "brief explanation"
}}

Be conservative with confidence scores. Use 0.8+ only when brand and model are clearly identified."""

    start_time = time.time()

    try:
        message = anthropic.messages.create(
            model='claude-haiku-4-5-20251001',  # Verified model ID
            max_tokens=512,
            temperature=0.0,
            messages=[{
                'role': 'user',
                'content': prompt
            }]
        )

        latency = (time.time() - start_time) * 1000

        # Extract JSON from response
        response_text = message.content[0].text
        json_match = response_text[response_text.find('{'):response_text.rfind('}')+1]
        parsed_data = json.loads(json_match)

        # Calculate cost ($1 input / $5 output per million tokens)
        input_cost = (message.usage.input_tokens / 1_000_000) * 1.0
        output_cost = (message.usage.output_tokens / 1_000_000) * 5.0
        cost = input_cost + output_cost

        total_cost += cost
        claude_call_count += 1

        print(f"[Claude] ✅ Parsed: {parsed_data['brand']} {parsed_data['model']} ({latency:.0f}ms)")
        print(f"   Tokens: {message.usage.input_tokens} in + {message.usage.output_tokens} out = ${cost:.6f}")
        print(f"   API calls: {claude_call_count}/{MAX_CLAUDE_CALLS}")

        parsed_data['latency'] = latency
        parsed_data['cost'] = cost
        parsed_data['tokens_used'] = {
            'input': message.usage.input_tokens,
            'output': message.usage.output_tokens,
            'total': message.usage.input_tokens + message.usage.output_tokens
        }

        return parsed_data

    except Exception as e:
        latency = (time.time() - start_time) * 1000
        print(f"[Claude] ❌ Exception: {e}")
        return {
            'brand': None,
            'model': None,
            'variant': None,
            'estimatedValue': None,
            'confidence': 0.0,
            'error': str(e),
            'latency': latency,
            'cost': 0.001  # Estimate
        }

print("✅ Claude Haiku parser ready")
```

```python
def validate_item(item: Dict) -> Dict:
    """
    Run full Layer 2b validation for a single item

    Workflow:
    1. Try barcode lookup (if barcode present)
    2. Fallback to visual search (if barcode fails or not present)
    3. Parse visual matches with Claude
    4. Calculate accuracy vs ground truth

    Args:
        item: Item dict from golden dataset manifest

    Returns:
        Dict with results, accuracy scores, costs, latency
    """
    item_id = item['id']
    has_barcode = item['has_barcode']
    ground_truth = item['ground_truth']

    print(f"\n{'='*80}")
    print(f"Validating: {item_id} (barcode: {has_barcode})")
    print(f"{'='*80}")

    result = {
        'item_id': item_id,
        'category': item['category'],
        'has_barcode': has_barcode,
        'ground_truth': ground_truth,
        'barcode_lookup': None,
        'visual_search': None,
        'llm_parsing': None,
        'accuracy': {},
        'total_latency': 0,
        'total_cost': 0.0,
        'path_taken': None
    }

    start_time = time.time()

    # STEP 1: Barcode lookup (if barcode present)
    if has_barcode and ground_truth['barcode']:
        barcode_value = ground_truth['barcode']['value']
        barcode_result = lookup_barcode(barcode_value, item_id)
        result['barcode_lookup'] = barcode_result
        result['total_cost'] += barcode_result['cost']

        # If barcode found, skip visual search (cost optimization)
        if barcode_result['found']:
            print(f"✅ Barcode match found - skipping visual search")
            result['path_taken'] = 'barcode_only'
            result['total_latency'] = barcode_result['latency']

            # Calculate accuracy
            predicted = barcode_result['product']
            actual = ground_truth['product']

            result['accuracy'] = {
                'brand_match': (predicted.get('brand', '').lower() == actual['brand'].lower()),
                'name_match': (actual['name'].lower() in predicted.get('name', '').lower()),
                'overall_correct': True  # Barcode is authoritative
            }

            return result

    # STEP 2: Visual search (barcode not present or not found)
    # Note: For validation, we need public image URLs
    # In real implementation, upload to GCS and get signed URL
    image_url = item.get('public_image_url')  # Must be provided in manifest

    if not image_url:
        print(f"⚠️ No public image URL - cannot run visual search")
        result['path_taken'] = 'failed_no_image'
        result['total_latency'] = time.time() - start_time
        return result

    serpapi_result = search_with_serpapi(image_url, item_id)
    result['visual_search'] = serpapi_result
    result['total_cost'] += serpapi_result['cost']
    result['path_taken'] = 'barcode_fallback' if has_barcode else 'visual_only'

    # STEP 3: Parse visual matches with Claude
    if serpapi_result.get('visual_matches'):
        claude_result = parse_serpapi_results(serpapi_result['visual_matches'], item_id)
        result['llm_parsing'] = claude_result
        result['total_cost'] += claude_result['cost']

        # Calculate accuracy
        predicted = claude_result
        actual = ground_truth['product']

        brand_match = (predicted.get('brand', '').lower() == actual['brand'].lower()) if predicted.get('brand') else False
        model_match = (predicted.get('model', '').lower() == actual['model'].lower()) if predicted.get('model') else False

        # Price accuracy (within 20%)
        price_match = False
        if predicted.get('estimatedValue') and actual['estimatedValue']:
            price_diff = abs(predicted['estimatedValue'] - actual['estimatedValue']) / actual['estimatedValue']
            price_match = price_diff <= 0.20

        result['accuracy'] = {
            'brand_match': brand_match,
            'model_match': model_match,
            'price_match': price_match,
            'overall_correct': brand_match and model_match,
            'confidence': predicted.get('confidence', 0.0)
        }
    else:
        print(f"⚠️ No visual matches found")
        result['accuracy'] = {
            'brand_match': False,
            'model_match': False,
            'price_match': False,
            'overall_correct': False,
            'confidence': 0.0
        }

    result['total_latency'] = (time.time() - start_time) * 1000

    print(f"\n📊 Results:")
    print(f"   Path: {result['path_taken']}")
    print(f"   Accuracy: {result['accuracy']}")
    print(f"   Latency: {result['total_latency']:.0f}ms")
    print(f"   Cost: ${result['total_cost']:.6f}")

    return result

print("✅ Validation orchestration ready")
```

```python
        sample_size: If provided, randomly sample N items (for budget control)

    Returns:
        DataFrame with all validation results
    """
    # Load manifest
    with open(manifest_path, 'r') as f:
        manifest = json.load(f)

    items = manifest['items']

    # Sample if requested
    if sample_size and sample_size < len(items):
        import random
        items = random.sample(items, sample_size)
        print(f"📦 Sampling {sample_size} items from {len(manifest['items'])} total")

    print(f"🚀 Starting validation of {len(items)} items...")
    print(f"   Budget: {MAX_SERPAPI_CALLS} SerpAPI calls, {MAX_CLAUDE_CALLS} Claude calls")

    results = []

    for i, item in enumerate(items, 1):
        print(f"\n[{i}/{len(items)}]", end=' ')

        try:
            result = validate_item(item)
            results.append(result)

            # Save intermediate results every 10 items
            if i % 10 == 0:
                df = pd.DataFrame(results)
                df.to_csv('results/layer2b_accuracy_intermediate.csv', index=False)
                print(f"\n💾 Saved intermediate results ({i} items)")

        except Exception as e:
            print(f"\n❌ Failed to validate {item['id']}: {e}")
            results.append({
                'item_id': item['id'],
                'category': item['category'],
                'error': str(e),
                'total_cost': 0.0
            })

    # Convert to DataFrame
    df = pd.DataFrame(results)

    # Save final results
    df.to_csv('results/layer2b_accuracy.csv', index=False)
    print(f"\n✅ Validation complete!")
    print(f"   Results saved: results/layer2b_accuracy.csv")
    print(f"   Total cost: ${total_cost:.2f}")
    print(f"   SerpAPI calls: {serpapi_call_count}/{MAX_SERPAPI_CALLS}")
    print(f"   Claude calls: {claude_call_count}/{MAX_CLAUDE_CALLS}")

    return df

print("✅ Batch validation ready")
```

```python
def analyze_accuracy(df: pd.DataFrame) -> Dict:
    """
    Calculate aggregate accuracy metrics from validation results

    Args:
        df: DataFrame with validation results

    Returns:
        Dict with all accuracy metrics
    """
    print("📊 Calculating accuracy metrics...")

    # Filter out errors
    valid_results = df[df['accuracy'].notna()].copy()

    # Extract accuracy fields
    valid_results['brand_match'] = valid_results['accuracy'].apply(lambda x: x.get('brand_match', False) if isinstance(x, dict) else False)
    valid_results['model_match'] = valid_results['accuracy'].apply(lambda x: x.get('model_match', False) if isinstance(x, dict) else False)
    valid_results['overall_correct'] = valid_results['accuracy'].apply(lambda x: x.get('overall_correct', False) if isinstance(x, dict) else False)

    # Calculate metrics
    metrics = {
        'total_items': len(df),
        'valid_results': len(valid_results),
        'errors': len(df) - len(valid_results),

        # Barcode metrics
        'barcode_items': len(df[df['has_barcode'] == True]),
        'barcode_found_rate': len(df[(df['has_barcode'] == True) & (df['barcode_lookup'].apply(lambda x: x.get('found', False) if isinstance(x, dict) else False))]) / max(len(df[df['has_barcode'] == True]), 1),

        # Visual search metrics
        'visual_search_items': len(df[df['visual_search'].notna()]),
        'visual_search_accuracy': valid_results[valid_results['visual_search'].notna()]['overall_correct'].mean() if len(valid_results[valid_results['visual_search'].notna()]) > 0 else 0.0,

        # LLM parsing metrics
        'llm_parsing_items': len(df[df['llm_parsing'].notna()]),
        'llm_brand_accuracy': valid_results['brand_match'].mean() if len(valid_results) > 0 else 0.0,
        'llm_model_accuracy': valid_results['model_match'].mean() if len(valid_results) > 0 else 0.0,
        'llm_overall_accuracy': valid_results['overall_correct'].mean() if len(valid_results) > 0 else 0.0,

        # Latency metrics
        'avg_latency_ms': df['total_latency'].mean(),
        'p95_latency_ms': df['total_latency'].quantile(0.95),

        # Cost metrics
        'total_cost': df['total_cost'].sum(),
        'avg_cost_per_item': df['total_cost'].mean(),

        # Path distribution
        'barcode_only_items': len(df[df['path_taken'] == 'barcode_only']),
        'visual_only_items': len(df[df['path_taken'] == 'visual_only']),
        'barcode_fallback_items': len(df[df['path_taken'] == 'barcode_fallback'])
    }

    # Calculate cost savings
    visual_only_cost = metrics['total_items'] * 0.016  # SerpAPI + Claude per item
    actual_cost = metrics['total_cost']
    metrics['cost_savings_dollars'] = visual_only_cost - actual_cost
    metrics['cost_savings_percent'] = (metrics['cost_savings_dollars'] / visual_only_cost) * 100 if visual_only_cost > 0 else 0.0

    return metrics

print("✅ Accuracy analysis ready")
```
