**Step 3: Update the trigger to use defineSecret pattern**

Modify `functions/src/triggers/onItemCreatedGemini3.ts`:

```typescript
import { onDocumentCreated } from 'firebase-functions/v2/firestore';
import { defineSecret } from 'firebase-functions/params';
import { handleItemCreated } from '../ai-pipeline/gemini/orchestrator';

// Define secrets for 2nd gen Cloud Functions
const googleApiKey = defineSecret('GOOGLE_API_KEY');
const serpApiKey = defineSecret('SERPAPI_KEY');

/**
 * Firestore trigger that fires when a new item is created.
 * Processes the item through the Gemini 3 Pro AI pipeline.
 *
 * Secrets are bound at deploy time and accessed via .value() at runtime.
 */
export const onItemCreatedGemini3 = onDocumentCreated(
  {
    document: 'items/{itemId}',
    secrets: [googleApiKey, serpApiKey],
    region: 'us-central1',
    memory: '512MiB',
    timeoutSeconds: 120,
  },
  async (event) => {
    if (!event.data) {
      console.error('No data in event');
      return;
    }

    // Secrets are available via .value() at runtime
    process.env.GOOGLE_API_KEY = googleApiKey.value();
    process.env.SERPAPI_KEY = serpApiKey.value();

    await handleItemCreated(event.data, {
      params: { itemId: event.params.itemId },
    });
  }
);
```

**Step 1: Create integration test setup file**

Create `functions/src/ai-pipeline/__tests__/integration-setup.ts`:

```typescript
/**
 * Integration test setup for Gemini 3 Pro pipeline.
 *
 * IMPORTANT: Integration tests require:
 * 1. Real Firebase Storage (emulator doesn't support getSignedUrl)
 * 2. Valid API keys in environment variables
 * 3. Service account with storage.objectViewer and serviceAccountTokenCreator roles
 *
 * Run with: npm run test:integration
 */

import * as admin from 'firebase-admin';
import * as path from 'path';

// Check for required environment variables
const requiredEnvVars = ['GOOGLE_API_KEY', 'SERPAPI_KEY', 'FIREBASE_PROJECT_ID'];

export function validateIntegrationEnvironment(): void {
  const missing = requiredEnvVars.filter((v) => !process.env[v]);
  if (missing.length > 0) {
    throw new Error(
      `Integration tests require environment variables: ${missing.join(', ')}\n` +
      'Create a .env.integration file or set them in your shell.'
    );
  }
}

export function initializeFirebaseAdmin(): admin.app.App {
  // Use application default credentials for GCP environment
  // Or service account JSON for local development
  const serviceAccountPath = process.env.GOOGLE_APPLICATION_CREDENTIALS;

  if (admin.apps.length > 0) {
    return admin.apps[0]!;
  }

  if (serviceAccountPath) {
    const serviceAccount = require(path.resolve(serviceAccountPath));
    return admin.initializeApp({
      credential: admin.credential.cert(serviceAccount),
      storageBucket: `${process.env.FIREBASE_PROJECT_ID}.firebasestorage.app`,
    });
  }

  // Use ADC in GCP environment
  return admin.initializeApp({
    storageBucket: `${process.env.FIREBASE_PROJECT_ID}.firebasestorage.app`,
  });
}

export async function uploadTestImage(
  bucket: admin.storage.Bucket,
  localPath: string,
  destinationPath: string
): Promise<string> {
  await bucket.upload(localPath, {
    destination: destinationPath,
    metadata: {
      contentType: 'image/jpeg',
    },
  });

  const file = bucket.file(destinationPath);
  const [signedUrl] = await file.getSignedUrl({
    action: 'read',
    expires: Date.now() + 60 * 60 * 1000, // 1 hour
  });

  return signedUrl;
}

export async function cleanupTestImage(
  bucket: admin.storage.Bucket,
  destinationPath: string
): Promise<void> {
  try {
    await bucket.file(destinationPath).delete();
  } catch (error) {
    // Ignore if file doesn't exist
    console.log(`Cleanup: ${destinationPath} not found, skipping`);
  }
}
```

**Step 1: Create the integration test**

Create `functions/src/ai-pipeline/gemini/__tests__/gemini-pipeline.integration.test.ts`:

```typescript
/**
 * Integration test for Gemini 3 Pro pipeline.
 *
 * Requirements:
 * - GOOGLE_API_KEY: Valid Google AI Studio API key
 * - SERPAPI_KEY: Valid SerpAPI key (Developer plan)
 * - FIREBASE_PROJECT_ID: Firebase project with Storage enabled
 * - GOOGLE_APPLICATION_CREDENTIALS: Path to service account JSON (for local dev)
 *
 * Run: npm run test:integration
 */

import * as admin from 'firebase-admin';
import { processItemWithGemini } from '../gemini-service';
import { validateCatalogItem } from '../schemas/catalog-item';
import {
  validateIntegrationEnvironment,
  initializeFirebaseAdmin,
  uploadTestImage,
  cleanupTestImage,
} from '../../__tests__/integration-setup';

// Skip if running in CI without credentials
const describeIntegration = process.env.GOOGLE_API_KEY ? describe : describe.skip;

describeIntegration('Gemini 3 Pro Pipeline Integration', () => {
  let app: admin.app.App;
  let bucket: admin.storage.Bucket;
  const testImagePath = 'test-items/integration-test-item.jpg';

  beforeAll(async () => {
    validateIntegrationEnvironment();
    app = initializeFirebaseAdmin();
    bucket = admin.storage().bucket();
  });

  afterAll(async () => {
    await cleanupTestImage(bucket, testImagePath);
    await app.delete();
  });

  describe('processItemWithGemini', () => {
    it('should analyze a product image and return valid CatalogItem', async () => {
      // Upload test image to Firebase Storage
      // Using a publicly available product image for testing
      const testImageUrl = 'https://storage.googleapis.com/abundance-test-public/test-product.jpg';

      // Process with Gemini
      const result = await processItemWithGemini(testImageUrl, undefined);

      // Validate result structure
      expect(result).toBeDefined();
      expect(result.name).toBeDefined();
      expect(result.category).toBeDefined();

      // Validate against schema
      const validation = validateCatalogItem(result);
      expect(validation.isValid).toBe(true);

      // Log for manual inspection
      console.log('Integration test result:', JSON.stringify(result, null, 2));
    }, 60000); // 60 second timeout

    it('should handle barcode lookup when barcode provided', async () => {
      const testImageUrl = 'https://storage.googleapis.com/abundance-test-public/test-product.jpg';
      const testBarcode = '0012345678905'; // Test UPC

      const result = await processItemWithGemini(testImageUrl, testBarcode);

      expect(result).toBeDefined();
      // Barcode should trigger barcode_lookup tool call
      console.log('Barcode test result:', JSON.stringify(result, null, 2));
    }, 60000);
  });

  describe('Cost Tracking', () => {
    it('should track API costs within expected range', async () => {
      const testImageUrl = 'https://storage.googleapis.com/abundance-test-public/test-product.jpg';

      const startTime = Date.now();
      const result = await processItemWithGemini(testImageUrl, undefined);
      const duration = Date.now() - startTime;

      // Verify timing is within acceptable range (< 30 seconds)
      expect(duration).toBeLessThan(30000);

      // Log timing for cost estimation
      console.log(`Processing time: ${duration}ms`);
      console.log(`Estimated cost: $0.03-0.04 per item`);
    }, 60000);
  });
});
```

## Troubleshooting

### Secret Access Errors

```
Error: Secret version not found or access denied
```

**Solution**: Verify secrets exist and function has access:
```bash
firebase functions:secrets:access GOOGLE_API_KEY
```

### Timeout Errors

```

**Solution**: Increase timeout in trigger config or optimize tool calls.

### Signed URL Errors

```
Error: Cannot sign data without 'client_email'
```

**Solution**: Ensure service account has `serviceAccountTokenCreator` role.
