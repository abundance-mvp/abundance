# Cell 4: Gemini Attribute Extraction Function

def extract_attributes_gemini(image_path: Path, item_id: str) -> Dict[str, Any]:
    """
    Extract attributes from image using Gemini 2.5 Flash-Lite.

    Args:
        image_path: Path to image file
        item_id: Item identifier for logging

    Returns:
        Dictionary with extracted attributes + metadata
    """
    try:
        # Initialize client
        client = genai.Client(api_key=GOOGLE_API_KEY)

        # Read image
        with open(image_path, 'rb') as f:
            image_bytes = f.read()

        # Generate content with JSON Schema Mode
        start_time = datetime.now()

        response = client.models.generate_content(
            model='gemini-2.5-flash-lite',
            contents=[
                EXTRACTION_PROMPT,
                genai.types.Part.from_bytes(
                    data=image_bytes,
                    mime_type='image/jpeg'
                )
            ],
            config=genai.types.GenerateContentConfig(
                temperature=0.2,  # Low temperature for consistent extraction
                top_p=0.8,
                top_k=40,
                max_output_tokens=256,
                response_mime_type='application/json',
                response_schema=ATTRIBUTE_SCHEMA
            )
        )

        latency = (datetime.now() - start_time).total_seconds() * 1000  # ms

        # Parse response
        attributes = json.loads(response.text)

        # Add metadata
        attributes['model'] = 'gemini-2.5-flash-lite'
        attributes['latency_ms'] = latency
        attributes['input_tokens'] = response.usage_metadata.prompt_token_count
        attributes['output_tokens'] = response.usage_metadata.candidates_token_count
        attributes['total_tokens'] = response.usage_metadata.total_token_count

        # Calculate cost
        input_cost = attributes['input_tokens'] * INPUT_TOKEN_PRICE
        output_cost = attributes['output_tokens'] * OUTPUT_TOKEN_PRICE
        attributes['cost'] = input_cost + output_cost

        return attributes

    except Exception as e:
        print(f"❌ Error extracting attributes for {item_id}: {e}")
        return {
            'error': str(e),
            'item_id': item_id
        }

print("✅ Gemini extraction function defined")

# Cell 7: Batch Attribute Extraction

def run_batch_extraction(items: List[Dict], max_cost: float = MAX_VALIDATION_COST) -> List[Dict]:
    """
    Run attribute extraction on all items with cost tracking.

    Args:
        items: List of dataset items
        max_cost: Maximum allowed cost (budget cap)

    Returns:
        List of results with predictions and ground truth
    """
    results = []
    total_cost = 0.0

    print(f"Starting batch extraction on {len(items)} items...")
    print(f"Budget cap: ${max_cost:.2f}\n")

    for i, item in enumerate(items, 1):
        # Check budget
        if total_cost >= max_cost:
            print(f"\n⚠️ Budget cap reached (${total_cost:.4f}), stopping at item {i-1}/{len(items)}")
            break

        # Extract attributes
        image_path = IMAGES_DIR / item['image_filename']

        if not image_path.exists():
            print(f"⚠️ [{i}/{len(items)}] Image not found: {image_path}")
            results.append({
                'item_id': item['id'],
                'error': 'Image not found',
                'ground_truth': item['ground_truth']
            })
            continue

        print(f"[{i}/{len(items)}] Processing {item['id']}...", end=' ')

        prediction = extract_attributes_gemini(image_path, item['id'])
        total_cost += prediction.get('cost', 0)

        # Store result
        result = {
            'item_id': item['id'],
            'description': item['description'],
            'ground_truth': item['ground_truth'],
            'prediction': prediction
        }
        results.append(result)

        # Calculate accuracy for this item
        gt = item['ground_truth']
        category_correct = prediction.get('category') == gt['category']
        color_correct = prediction.get('color') == gt['color']
        condition_correct = prediction.get('condition') == gt['condition']

        status = '✅' if (category_correct and color_correct and condition_correct) else '⚠️'
        print(f"{status} {prediction.get('latency_ms', 0):.0f}ms, ${prediction.get('cost', 0):.6f}")

    print(f"\n✅ Batch extraction complete")
    print(f"   Items processed: {len(results)}")
    print(f"   Total cost: ${total_cost:.4f}")
    print(f"   Average cost per item: ${total_cost / len(results):.6f}")

    return results

# Run batch extraction
batch_results = run_batch_extraction(items)

# Cell 9: Accuracy Calculation Functions

def normalize_color(color: str) -> str:
    """Normalize color names for comparison."""
    color_map = {
        'navy': 'blue',
        'dark blue': 'blue',
        'light blue': 'blue',
        'sky blue': 'blue',
        'dark green': 'green',
        'light green': 'green',
        'dark red': 'red',
        'light red': 'red',
        'off-white': 'white',
        'cream': 'white',
        'beige': 'brown',
        'tan': 'brown',
        'silver': 'gray',
        'grey': 'gray',
    }
    normalized = color.lower().strip()
    return color_map.get(normalized, normalized)

def is_condition_within_1(predicted: str, actual: str) -> bool:
    """Check if condition prediction within ±1 level."""
    condition_scale = {
        'new': 5,
        'like-new': 4,
        'good': 3,
        'fair': 2,
        'poor': 1
    }

    pred_val = condition_scale.get(predicted)
    actual_val = condition_scale.get(actual)

    if pred_val is None or actual_val is None:
        return False

    return abs(pred_val - actual_val) <= 1

def calculate_accuracy_metrics(results: List[Dict]) -> Dict[str, Any]:
    """
    Calculate accuracy metrics for all attributes.

    Returns dictionary with:
    - category_accuracy, color_accuracy, material_accuracy, condition_exact, condition_within_1
    - per-category precision, recall, F1
    - confusion matrix
    - average latency, total cost
    """
    valid_results = [r for r in results if 'error' not in r['prediction']]

    if not valid_results:
        return {'error': 'No valid results to calculate metrics'}

    # Initialize counters
    category_correct = 0
    color_correct = 0
    material_correct = 0
    condition_exact = 0
    condition_within_1_count = 0

    total_latency = 0
    total_cost = 0
    total_tokens = 0

    # Confusion matrix (category)
    categories = ['camping', 'electronics', 'home-decor', 'clothing', 'kitchenware', 'books', 'toys', 'sports', 'tools', 'other']
    confusion_matrix = {cat: {cat2: 0 for cat2 in categories} for cat in categories}

    # Process each result
    for result in valid_results:
        gt = result['ground_truth']
        pred = result['prediction']

        # Category
        if pred.get('category') == gt['category']:
            category_correct += 1
        confusion_matrix[gt['category']][pred.get('category', 'other')] += 1

        # Color (normalized)
        if normalize_color(pred.get('color', '')) == normalize_color(gt['color']):
            color_correct += 1

        # Material (if present in ground truth)
        if gt.get('material'):
            if pred.get('material', '').lower() == gt['material'].lower():
                material_correct += 1

        # Condition
        if pred.get('condition') == gt['condition']:
            condition_exact += 1
        if is_condition_within_1(pred.get('condition', ''), gt['condition']):
            condition_within_1_count += 1

        # Metadata
        total_latency += pred.get('latency_ms', 0)
        total_cost += pred.get('cost', 0)
        total_tokens += pred.get('total_tokens', 0)

    n = len(valid_results)

    return {
        'total_items': n,
        'category_accuracy': (category_correct / n) * 100,
        'color_accuracy': (color_correct / n) * 100,
        'material_accuracy': (material_correct / n) * 100,  # Note: denominator should be items with material GT
        'condition_exact': (condition_exact / n) * 100,
        'condition_within_1': (condition_within_1_count / n) * 100,
        'avg_latency_ms': total_latency / n,
        'total_cost': total_cost,
        'avg_cost_per_item': total_cost / n,
        'avg_tokens_per_item': total_tokens / n,
        'confusion_matrix': confusion_matrix
    }

print("✅ Accuracy calculation functions defined")

def calculate_per_category_accuracy(results: List[Dict]) -> pd.DataFrame:
    """Calculate accuracy metrics per category."""
    category_stats = []

    categories = ['camping', 'electronics', 'home-decor', 'clothing', 'kitchenware', 'books', 'toys', 'sports', 'tools', 'other']

    for category in categories:
        category_results = [r for r in results if r['ground_truth']['category'] == category and 'error' not in r['prediction']]

        if not category_results:
            continue

        n = len(category_results)
        category_correct = sum(1 for r in category_results if r['prediction'].get('category') == r['ground_truth']['category'])
        color_correct = sum(1 for r in category_results if normalize_color(r['prediction'].get('color', '')) == normalize_color(r['ground_truth']['color']))
        condition_correct = sum(1 for r in category_results if r['prediction'].get('condition') == r['ground_truth']['condition'])

        category_stats.append({
            'Category': category,
            'Count': n,
            'Category Acc (%)': (category_correct / n) * 100,
            'Color Acc (%)': (color_correct / n) * 100,
            'Condition Acc (%)': (condition_correct / n) * 100
        })

    return pd.DataFrame(category_stats)
