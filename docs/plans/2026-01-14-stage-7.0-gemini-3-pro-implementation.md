```typescript
// functions/src/ai-pipeline/gemini/schemas/catalog-item.ts

/**
 * Condition enum for item condition assessment
 */
export type Condition = 'new' | 'like-new' | 'good' | 'fair' | 'poor';

/**
 * Confidence level for catalog identification
 */
export type Confidence = 'high' | 'medium' | 'low';

/**
 * CatalogItem - the output schema for Gemini 3 Pro
 *
 * Used for:
 * 1. TypeScript type checking
 * 2. Gemini responseSchema for structured output
 * 3. Firestore document structure
 */
export interface CatalogItem {
  // Core identification
  name: string;                    // "Nike Air Max 90"
  category: string;                // "footwear" (free-form)
  subCategory: string;             // "sneakers" (free-form)

  // Product details
  brand: string | null;            // "Nike"
  model: string | null;            // "Air Max 90"
  color: string;                   // "white/red"
  condition: Condition;            // "good"
  dimensions: string | null;       // "size 10 mens", or "barcode: 012345" if lookup failed
  quantity: number;                // 1 (or more if multiple identical items)

  // Valuation
  estimatedValue: number | null;   // 85.00 (per-unit, condition-adjusted)
  confidence: Confidence;          // "high"
}

/**
 * Validation result
 */
export interface ValidationResult {
  valid: boolean;
  errors: string[];
}

/**
 * Valid condition values
 */
const VALID_CONDITIONS: Condition[] = ['new', 'like-new', 'good', 'fair', 'poor'];

/**
 * Valid confidence values
 */
const VALID_CONFIDENCES: Confidence[] = ['high', 'medium', 'low'];

/**
 * Validate a CatalogItem object
 */
export function validateCatalogItem(item: CatalogItem): ValidationResult {
  const errors: string[] = [];

  // Required string fields
  if (!item.name || typeof item.name !== 'string') {
    errors.push('Name is required and must be a string');
  }
  if (!item.category || typeof item.category !== 'string') {
    errors.push('Category is required and must be a string');
  }
  if (!item.subCategory || typeof item.subCategory !== 'string') {
    errors.push('SubCategory is required and must be a string');
  }
  if (!item.color || typeof item.color !== 'string') {
    errors.push('Color is required and must be a string');
  }

  // Condition validation
  if (!VALID_CONDITIONS.includes(item.condition)) {
    errors.push(`Invalid condition: ${item.condition}`);
  }

  // Confidence validation
  if (!VALID_CONFIDENCES.includes(item.confidence)) {
    errors.push(`Invalid confidence: ${item.confidence}`);
  }

  // Quantity validation
  if (typeof item.quantity !== 'number' || item.quantity < 1) {
    errors.push('Quantity must be positive');
  }

  // Optional numeric field validation
  if (item.estimatedValue !== null && typeof item.estimatedValue !== 'number') {
    errors.push('EstimatedValue must be a number or null');
  }

  return {
    valid: errors.length === 0,
    errors
  };
}

/**
 * JSON Schema for Gemini responseSchema
 * Used with responseMimeType: "application/json"
 */
export const CATALOG_ITEM_SCHEMA = {
  type: 'object',
  properties: {
    name: {
      type: 'string',
      description: 'Product name, e.g., "Nike Air Max 90"'

    category: {
      type: 'string',
      description: 'Primary category, e.g., "footwear", "electronics", "furniture"'
    },
    subCategory: {

    },
    brand: {
      type: ['string', 'null'],
      description: 'Brand name if identifiable, null otherwise'
    },
    model: {
      type: ['string', 'null'],
      description: 'Model name/number if identifiable, null otherwise'
    },
    color: {
      type: 'string',
      description: 'Primary color(s), e.g., "white/red", "black"'
    },
    condition: {
      type: 'string',
      enum: ['new', 'like-new', 'good', 'fair', 'poor'],
      description: 'Physical condition assessment'
    },

      type: ['string', 'null'],

      minimum: 1,
```

```typescript
// functions/src/ai-pipeline/tools/google-lens.ts
import fetch from 'node-fetch';

/**
 * Google Lens search result
 */
export interface GoogleLensResult {
  exact_matches: boolean;
  products: Array<{
    title: string;
    brand?: string;
    price?: number;
    source: string;
    link: string;
  }>;
}

/**
 * Search for product information using SerpAPI Google Lens
 *
 * @param imageUrl - Public URL of the product image
 * @returns GoogleLensResult with visual matches
 */
export async function searchGoogleLens(imageUrl: string): Promise<GoogleLensResult> {
  const apiKey = process.env.SERPAPI_KEY;

  if (!apiKey) {
    throw new Error('SERPAPI_KEY environment variable is required');
  }

  const url = new URL('https://serpapi.com/search');
  url.searchParams.set('engine', 'google_lens');
  url.searchParams.set('url', imageUrl);
  url.searchParams.set('api_key', apiKey);

  const response = await fetch(url.toString(), {
    method: 'GET',
    headers: {
      'Accept': 'application/json'
    },
    timeout: 30000
  });

  if (!response.ok) {
    throw new Error(`SerpAPI request failed: ${response.status} ${response.statusText}`);
  }

  const data = await response.json() as any;
  const visualMatches = data.visual_matches || [];

  // Check if any match is marked as exact
  const hasExactMatch = visualMatches.some((m: any) => m.exact_match === true);

  // Extract top 5 product matches
  const products = visualMatches.slice(0, 5).map((match: any) => ({
    title: match.title || 'Unknown Product',
    brand: extractBrand(match),
    price: match.price?.extracted_value,
    source: match.source || 'Unknown',
    link: match.link || ''
  }));

  return {
    exact_matches: hasExactMatch,
    products
  };
}

/**
 * Extract brand from match data
 * Brand may be in dedicated field, title, or source
 */
function extractBrand(match: any): string | undefined {
  // Direct brand field (rare)
  if (match.brand) {
    return match.brand;
  }

  // Extract from title (e.g., "Nike Air Max 90" -> "Nike")
  const title = match.title || '';
  const knownBrands = [
    'Nike', 'Adidas', 'Apple', 'Samsung', 'Sony', 'LG', 'Dell', 'HP',
    'Coleman', 'REI', 'Patagonia', 'North Face', 'IKEA', 'Cuisinart',
    'KitchenAid', 'Dyson', 'Bose', 'JBL', 'Canon', 'Nikon'
  ];

  for (const brand of knownBrands) {
    if (title.toLowerCase().includes(brand.toLowerCase())) {
      return brand;
    }
  }

  return undefined;
}
```

```typescript
 * Search for product pricing using Google Search Grounding
 *
 * Uses Gemini with Google Search to find current market prices
 * Cost: $0.014 per query ($14/1K queries)
 *
 * @param query - Search query, e.g., "Nike Air Max 90 size 10 price"
 * @returns WebSearchResult with price data from e-commerce sites
 */
export async function searchWeb(query: string): Promise<WebSearchResult> {
  const apiKey = process.env.GOOGLE_API_KEY;

  if (!apiKey) {
    throw new Error('GOOGLE_API_KEY environment variable is required');
  }

  const ai = new GoogleGenAI({ apiKey });

  try {
    const response = await ai.models.generateContent({
      model: 'gemini-3-pro-preview',
      contents: `Find current market prices for: ${query}

Search e-commerce sites (Amazon, eBay, Walmart, Target) and return pricing data.

Return JSON in this exact format:
{
  "prices": [
    { "source": "Amazon", "price": 99.99, "condition": "new" },
    { "source": "eBay", "price": 75.00, "condition": "used" }
  ]
}

Only include prices you found. If no prices found, return {"prices": []}`,
      config: {
        tools: [{ googleSearchRetrieval: {} }],
        temperature: 0.1,
        maxOutputTokens: 512
      }
    });

    const text = response.text || '';

    try {
      // Try to parse JSON from response
      const jsonMatch = text.match(/\{[\s\S]*\}/);
      if (jsonMatch) {
        const parsed = JSON.parse(jsonMatch[0]);
        return {
          prices: parsed.prices || [],
        };
      }
      return { prices: [], raw: text };
    } catch {
      return { prices: [], raw: text };
    }

  } catch (error: any) {
```

```typescript
// functions/src/ai-pipeline/tools/tool-executor.ts
import { searchGoogleLens } from './google-lens';
import { lookupBarcode } from './barcode-lookup';
import { searchWeb } from './web-search';

/**
 * Execute a tool call from Gemini
 *
 * Routes function calls to their implementations:
 * - google_lens_search → searchGoogleLens
 * - barcode_lookup → lookupBarcode
 * - web_search → searchWeb
 *
 * @param name - Tool name from Gemini function call
 * @param args - Arguments from Gemini function call
 * @param imageUrl - Default image URL (for google_lens if not specified)
 * @returns Tool execution result
 */
export async function executeToolCall(
  name: string,
  args: Record<string, unknown>,
  imageUrl: string
): Promise<unknown> {
  switch (name) {
    case 'google_lens_search':
      // Use provided image_url or fall back to default
      const searchImageUrl = (args.image_url as string) || imageUrl;
      return searchGoogleLens(searchImageUrl);

    case 'barcode_lookup':
      return lookupBarcode(
        args.code as string,
        args.symbology as string | undefined
      );

    case 'web_search':
      return searchWeb(args.query as string);

    default:
      throw new Error(`Unknown tool: ${name}`);
  }
}
```

```typescript
// functions/src/ai-pipeline/gemini/gemini-service.ts
import { GoogleGenAI } from '@google/genai';
import fetch from 'node-fetch';
import { CatalogItem } from './schemas/catalog-item';
import { SYSTEM_PROMPT, CATALOG_TOOLS, GENERATION_CONFIG, GEMINI_MODEL_ID } from './prompts';
import { executeToolCall } from '../tools/tool-executor';

/**
 * Process an item image with Gemini 3 Pro
 *
 * Flow:
 * 1. Fetch image and convert to base64
 * 2. Send to Gemini with system prompt and tools
 * 3. Handle tool calls in a loop until final response
 * 4. Parse and return CatalogItem(s)
 *
 * @param imageUrl - Public URL of the cropped item image
 * @returns CatalogItem or array of CatalogItems for multi-object images
 */
export async function processItemWithGemini(
  imageUrl: string
): Promise<CatalogItem | CatalogItem[]> {
  const apiKey = process.env.GOOGLE_API_KEY;

  if (!apiKey) {
    throw new Error('GOOGLE_API_KEY environment variable is required');
  }

  // Fetch image and convert to base64
  const imageBase64 = await fetchImageBase64(imageUrl);

  const ai = new GoogleGenAI({ apiKey });

  // Build initial request
  let contents: any[] = [{
    role: 'user',
    parts: [
      { text: 'Analyze this image and create catalog entry(ies).' },
      {
        inlineData: {
          mimeType: 'image/jpeg',
          data: imageBase64
        }
      }
    ]
  }];

  // Flatten tool declarations for the API
  const toolDeclarations = CATALOG_TOOLS.flatMap(t => t.functionDeclarations || []);

  let response = await ai.models.generateContent({
    model: GEMINI_MODEL_ID,
    contents,
    config: {
      systemInstruction: SYSTEM_PROMPT,
      tools: [{ functionDeclarations: toolDeclarations }],
      ...GENERATION_CONFIG
    }
  });

  // Handle tool calls in a loop
  let iterations = 0;
  const maxIterations = 10; // Safety limit

  while (iterations < maxIterations) {
    const candidate = response.response?.candidates?.[0];
    const parts = candidate?.content?.parts || [];

    // Check for function calls
    const functionCalls = parts.filter((p: any) => p.functionCall);

    if (functionCalls.length === 0) {
      // No more tool calls - extract final result
      break;
    }

    // Execute all tool calls
    const toolResults = await Promise.all(
      functionCalls.map(async (part: any) => {
        const call = part.functionCall;
        const result = await executeToolCall(call.name, call.args || {}, imageUrl);
        return {
          functionResponse: {
            name: call.name,
            response: result
          }
        };
      })
    );

    // Add tool results to conversation
    contents = [
      ...contents,
      { role: 'model', parts },
      {
        role: 'user',
        parts: toolResults.map(r => ({
          functionResponse: r.functionResponse
        }))
      }
    ];

    // Continue conversation
    response = await ai.models.generateContent({
      model: GEMINI_MODEL_ID,
      contents,
      config: {
        systemInstruction: SYSTEM_PROMPT,
        tools: [{ functionDeclarations: toolDeclarations }],
        ...GENERATION_CONFIG
      }
    });

    iterations++;
  }

  // Extract final JSON response
  const finalCandidate = response.response?.candidates?.[0];
  const finalParts = finalCandidate?.content?.parts || [];
  const textPart = finalParts.find((p: any) => p.text);

  if (!textPart?.text) {
    throw new Error('No text response from Gemini');
  }

  return JSON.parse(textPart.text) as CatalogItem | CatalogItem[];
}

/**
 * Fetch image from URL and convert to base64
 */
async function fetchImageBase64(url: string): Promise<string> {
  const response = await fetch(url);

  if (!response.ok) {
    throw new Error(`Failed to fetch image: ${response.status} ${response.statusText}`);
  }

  const contentType = response.headers.get('content-type') || '';
  if (!contentType.startsWith('image/')) {
    throw new Error(`Invalid content type: expected image/*, got ${contentType}`);
  }

  const buffer = await response.arrayBuffer();
  return Buffer.from(buffer).toString('base64');
}
```

```typescript
// functions/src/ai-pipeline/gemini/orchestrator.ts
import * as admin from 'firebase-admin';
import { processItemWithGemini } from './gemini-service';
import { CatalogItem, validateCatalogItem } from './schemas/catalog-item';

/**
 * Handle item creation - process with Gemini 3 Pro
 *
 * Called by Firestore trigger when item document is created
 * Extracts signed URL, calls Gemini, updates document with results
 *
 * @param snapshot - Firestore document snapshot
 * @param context - Event context with params
 */
export async function handleItemCreated(
  snapshot: admin.firestore.DocumentSnapshot,
  context: { params: { itemId: string } }
): Promise<void> {
  const itemId = context.params.itemId;
  const itemData = snapshot.data();

  if (!itemData) {
    console.error(`[Gemini3] No data for item ${itemId}`);
    return;
  }

  // Skip if not in pending state
  if (itemData.status !== 'pending') {
    console.log(`[Gemini3] Skipping item ${itemId} (status: ${itemData.status})`);
    return;
  }

  console.log({
    severity: 'INFO',
    message: 'Gemini 3 Pro pipeline started',
    itemId,
    userId: itemData.userId
  });

  try {
    // Get signed URL for image
    const imageUrl = await getSignedImageUrl(itemData.imagePath);

    // Process with Gemini 3 Pro
    const result = await processItemWithGemini(imageUrl);

    // Validate result
    const catalogItems = Array.isArray(result) ? result : [result];
    for (const item of catalogItems) {
      const validation = validateCatalogItem(item);
      if (!validation.valid) {
        console.warn(`[Gemini3] Validation warnings for ${itemId}:`, validation.errors);
      }
    }

    // Update Firestore with results
    await snapshot.ref.update({
      status: 'complete',
      catalog: result,
      completedAt: admin.firestore.FieldValue.serverTimestamp(),
      updatedAt: admin.firestore.FieldValue.serverTimestamp()
    });

    console.log({
      severity: 'INFO',
      message: 'Gemini 3 Pro pipeline completed',
      itemId,
      itemCount: Array.isArray(result) ? result.length : 1
    });

  } catch (error: any) {
    console.error({
      severity: 'ERROR',
      message: 'Gemini 3 Pro pipeline failed',
      itemId,
      error: error.message
    });

    await snapshot.ref.update({
      status: 'failed',
      error: error.message,
      updatedAt: admin.firestore.FieldValue.serverTimestamp()
    });
  }
}

/**
 * Get signed URL for Cloud Storage image
 * Valid for 15 minutes
 */
async function getSignedImageUrl(imagePath: string): Promise<string> {
  const bucket = admin.storage().bucket();
  const file = bucket.file(imagePath);

  const [url] = await file.getSignedUrl({
    action: 'read',
    expires: Date.now() + 15 * 60 * 1000 // 15 minutes
  });

  return url;
}

/**
 * Log AI costs for monitoring
 */
export async function logCosts(
  itemId: string,
  toolsUsed: string[]
): Promise<void> {
  // Cost estimates based on COST-MODEL-001
  const toolCosts: Record<string, number> = {
    google_lens_search: 0.015,
    barcode_lookup: 0.005, // Amortized from monthly subscription
    web_search: 0.014
  };

  const geminiCost = 0.004; // ~$0.004 per item
  const totalToolCost = toolsUsed.reduce((sum, tool) => sum + (toolCosts[tool] || 0), 0);

  await admin.firestore().collection('aiCosts').add({
    itemId,
    timestamp: admin.firestore.FieldValue.serverTimestamp(),
    gemini: geminiCost,
    tools: Object.fromEntries(toolsUsed.map(t => [t, toolCosts[t] || 0])),
    total: geminiCost + totalToolCost
  });
}
```
