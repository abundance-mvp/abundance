```yaml
name: iOS Build Check

on:
  pull_request:
    paths:
      - 'ios/**'
      - '.github/workflows/ios-build-check.yml'
  push:
    branches: [main]

jobs:
  build:
    name: Build iOS App
    runs-on: macos-14

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Set up Xcode
        uses: maxim-lobanov/setup-xcode@v1
        with:
          xcode-version: '16.0'

      - name: Cache Swift packages
        uses: actions/cache@v4
        with:
          path: |
            ios/.build
            ~/Library/Developer/Xcode/DerivedData
          key: ${{ runner.os }}-spm-${{ hashFiles('**/Package.resolved') }}
          restore-keys: |
            ${{ runner.os }}-spm-

      - name: Build iOS project
        working-directory: ios
        run: swift build -c debug

      - name: Run SwiftLint
        working-directory: ios
        run: |
          if ! command -v swiftlint &> /dev/null; then
            brew install swiftlint
          fi
          swiftlint lint --strict

      - name: Run unit tests
        working-directory: ios
        run: swift test --parallel
```

### Task 4: GitHub Workflows - Spec Validation

**Files:**
- Create: `.github/workflows/spec-validation.yml`

**Step 1: Write spec validation workflow**

```yaml
name: Spec Validation

on:
  pull_request:
    paths:
      - 'docs/**'
      - '.github/workflows/spec-validation.yml'
  push:
    branches: [main]

jobs:
  validate:
    name: Validate Documentation
    runs-on: ubuntu-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Check for broken links
        run: |
          # Find all markdown files
          find docs -name "*.md" -type f | while read file; do
            echo "Checking $file for broken links..."
            # Extract markdown links [text](path)
            grep -oP '\[.*?\]\(\K[^)]+' "$file" | while read link; do
              # Skip external URLs
              if [[ $link == http* ]]; then
                continue
              fi
              # Resolve relative paths
              dir=$(dirname "$file")
              target="$dir/$link"
              if [[ ! -f "$target" ]]; then
                echo "ERROR: Broken link in $file: $link (resolved to $target)"
                exit 1
              fi
            done
          done

      - name: Validate ADR numbering
        run: |
          # Check for duplicate ADR numbers
          find docs/adr -name "ADR-*.md" -type f | \
            sed 's/.*ADR-\([0-9]\+\).*/\1/' | \
            sort | uniq -d | while read dup; do
              echo "ERROR: Duplicate ADR number: $dup"
              exit 1
            done

      - name: Check DESIGN doc cross-references
        run: |
          # Ensure all DESIGN docs reference at least one ADR
          find docs/design -name "DESIGN-*.md" -type f | while read file; do
            if ! grep -q "ADR-[0-9]\+" "$file"; then
              echo "WARNING: $file does not reference any ADRs"
            fi
          done
```

```yaml
name: Claude Code Action - CI Debugging

on:
  issue_comment:
    types: [created]

jobs:
  debug-ci:
    name: Debug CI Failures
    runs-on: ubuntu-latest

    # Only run when @claude is mentioned in PR comment
    if: |
      github.event.issue.pull_request &&
      contains(github.event.comment.body, '@claude')

    permissions:
      contents: read
      issues: write
      pull-requests: write

    steps:
      - name: Checkout PR code
        uses: actions/checkout@v4
        with:
          ref: ${{ github.event.issue.pull_request.head.ref }}

      - name: Get PR details
        id: pr
        uses: actions/github-script@v7
        with:
          script: |
            const pr = await github.rest.pulls.get({
              owner: context.repo.owner,
              repo: context.repo.repo,
              pull_number: context.issue.number
            });
            return pr.data;

      - name: Invoke Claude Code Action

          anthropic-api-key: ${{ secrets.ANTHROPIC_API_KEY }}
          github-token: ${{ secrets.GITHUB_TOKEN }}
          prompt: |
            A developer mentioned @claude in this PR comment.

            PR: ${{ steps.pr.outputs.title }}
            Comment: ${{ github.event.comment.body }}

            Please analyze recent CI failures and suggest fixes.
            Review the workflow logs and provide specific guidance.

          # CORRECTED: This is CI debugging via @claude mentions
          # NOT automatic CI failure response (ref: RESEARCH-VALIDATION-stage-5.3.md)
```

### Task 8: GitHub Workflows - Changelog Update

**Files:**
- Create: `.github/workflows/changelog-update.yml`

**Step 1: Write changelog update workflow**

```yaml
name: Changelog Update

on:
  release:
    types: [published]

jobs:
  update:
    name: Update CHANGELOG.md
    runs-on: ubuntu-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Generate changelog entry
        id: changelog
        run: |
          # Get commits since last release
          LAST_TAG=$(git describe --tags --abbrev=0 2>/dev/null || echo "")
          if [ -z "$LAST_TAG" ]; then
            COMMITS=$(git log --oneline)
          else
            COMMITS=$(git log ${LAST_TAG}..HEAD --oneline)
          fi

          # Format changelog entry
          echo "## [${{ github.event.release.tag_name }}] - $(date +%Y-%m-%d)" > /tmp/changelog.md
          echo "" >> /tmp/changelog.md
          echo "$COMMITS" | while read line; do
            echo "- $line" >> /tmp/changelog.md
          done
          echo "" >> /tmp/changelog.md

      - name: Update CHANGELOG.md
        run: |
          # Prepend new entry to CHANGELOG.md
          if [ -f CHANGELOG.md ]; then
            cat /tmp/changelog.md CHANGELOG.md > /tmp/new-changelog.md
            mv /tmp/new-changelog.md CHANGELOG.md
          else
            mv /tmp/changelog.md CHANGELOG.md
          fi

      - name: Commit changelog
        run: |
          git config user.name "github-actions[bot]"
          git config user.email "github-actions[bot]@users.noreply.github.com"
          git add CHANGELOG.md
          git commit -m "docs: update CHANGELOG.md for ${{ github.event.release.tag_name }}"
          git push
```

### Task 19: Claude Code Hook - Bash Command Validator

**Files:**
- Create: `.claude/hooks/bash_command_validator.py`

**Step 1: Write bash command validator**

```python
#!/usr/bin/env python3
"""
Bash Command Validator Hook
Blocks dangerous commands from Claude Code Bash tool
"""

import sys
import re

# Dangerous command patterns
DANGEROUS_PATTERNS = [
    r"rm\s+-rf\s+/",  # rm -rf / (recursive delete root)
    r"rm\s+-rf\s+\*",  # rm -rf * (delete all files)
    r":\(\)\{\s*:\|:&\s*\};:",  # Fork bomb
    r"dd\s+if=.*of=/dev/sd",  # Overwrite disk
    r"mkfs\.",  # Format filesystem
    r"curl.*\|\s*bash",  # Pipe to bash (unsafe)
    r"wget.*\|\s*sh",  # Pipe to sh (unsafe)
]

def validate_command(command: str) -> bool:
    """
    Validate a bash command for safety.
    Returns True if safe, False if dangerous.
    """
    for pattern in DANGEROUS_PATTERNS:
        if re.search(pattern, command, re.IGNORECASE):
            print(f"❌ BLOCKED: Dangerous command pattern detected")
            print(f"   Pattern: {pattern}")
            print(f"   Command: {command}")
            return False
    return True

def main():
    if len(sys.argv) < 2:
        print("Usage: bash_command_validator.py <command>")
        sys.exit(1)

    command = " ".join(sys.argv[1:])

    if validate_command(command):
        print("✅ Command validated as safe")
        sys.exit(0)
    else:
        sys.exit(1)

if __name__ == "__main__":
    main()
```
