export enum AIErrorType {
  // Retryable errors (exponential backoff)
  RATE_LIMIT = 'rate_limit',              // 429, RESOURCE_EXHAUSTED
  QUOTA_EXCEEDED = 'quota_exceeded',      // QUOTA_EXCEEDED

    public type: AIErrorType,

    public retryable: boolean,

**Retry Logic Template** (from DESIGN-042):
```typescript
export async function retryWithExponentialBackoff<T>(
  fn: () => Promise<T>,
  maxRetries: number = 5,
  errorHandler?: (error: AIError, attempt: number) => void
): Promise<T> {
  for (let attempt = 0; attempt < maxRetries; attempt++) {
    try {
      return await fn();
    } catch (error) {
      const aiError = wrapError(error);

      // Call custom error handler if provided
      if (errorHandler) {
        errorHandler(aiError, attempt);
      }

      // Don't retry non-retryable errors
      if (!aiError.retryable || attempt === maxRetries - 1) {
        throw aiError;
      }

      // Exponential backoff: 1s, 2s, 4s, 8s, 16s (capped at 60s)
      const delay = Math.min(Math.pow(2, attempt) * 1000, 60000);
      console.log(`[Retry] Attempt ${attempt + 1}/${maxRetries}, waiting ${delay}ms...`);
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }

  throw new Error('Max retries exceeded');
}

function wrapError(error: any): AIError {
  // Vertex AI errors
  if (error.code === 429 || error.message?.includes('RESOURCE_EXHAUSTED')) {
    return new AIError(AIErrorType.RATE_LIMIT, 'Vertex AI rate limit exceeded', true, error);
  }

  // Anthropic errors
  if (error.status === 429) {
    return new AIError(AIErrorType.RATE_LIMIT, 'Anthropic rate limit exceeded', true, error);
  }

  // SerpAPI errors
  if (error.message?.includes('exceeded your current quota')) {
    return new AIError(AIErrorType.QUOTA_EXCEEDED, 'SerpAPI quota exceeded', true, error);
  }

  // Default to non-retryable
  return new AIError(AIErrorType.TRANSIENT, error.message, false, error);
}
```

**Batch API Overview**:
Claude Sonnet 4.5 Batch API offers 50% cost discount ($1.50/$7.50 vs $3/$15) with up to 24h processing window.

**Batch Processing Pattern**:
```typescript
export interface BatchJobConfig {
  itemIds: string[];
  priority: 'high' | 'normal' | 'low';
  maxWaitTime: number; // milliseconds
}

export class BatchProcessor {
  private batchQueue: Map<string, BatchJobConfig> = new Map();

  async enqueueBatch(config: BatchJobConfig): Promise<string> {
    const batchId = generateBatchId();
    this.batchQueue.set(batchId, config);

    // Start batch processing after delay (to accumulate more items)
    setTimeout(() => this.processBatch(batchId), config.maxWaitTime);

    return batchId;
  }

  private async processBatch(batchId: string): Promise<void> {
    const config = this.batchQueue.get(batchId);
    if (!config) return;

    // Process all items in batch
    for (const itemId of config.itemIds) {
      await this.processItem(itemId);
    }

    this.batchQueue.delete(batchId);
  }

  private async processItem(itemId: string): Promise<void> {
    // Load Layer 2a + 2b data
    const item = await firestore.collection('items').doc(itemId).get();
    const layer2aData = item.data().layer2aData;
    const layer2bData = item.data().layer2bData;

    // Call Claude Sonnet 4.5 synthesis
    const anthropic = new AnthropicProvider(process.env.ANTHROPIC_API_KEY);
    const synthesis = await anthropic.synthesize({ layer2aData, layer2bData });

    // Update Firestore
    await firestore.collection('items').doc(itemId).update({
      layer3Data: synthesis,
      status: 'completed_layer3'
    });
  }
}
```
